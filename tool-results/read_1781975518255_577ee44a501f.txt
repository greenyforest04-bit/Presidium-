     1→"""Polls and Voting for Presidium — end-to-end encrypted group polls.
     2→
     3→Provides a complete polling system for group chats where all vote data
     4→is encrypted end-to-end. The poll creator defines questions and options,
     5→and group members cast encrypted votes that only the creator can decrypt
     6→and tally.
     7→
     8→Architecture:
     9→    PollManager
    10→        ├── Poll creation (question + options + settings)
    11→        ├── Vote casting (E2EE vote encryption)
    12→        ├── Vote tallying (creator-side decryption + counting)
    13→        ├── Poll closing (automatic or manual)
    14→        └── Results broadcast (encrypted summary to group)
    15→
    16→Privacy model:
    17→    - Poll definitions are encrypted with Sender Keys (group encryption)
    18→    - Individual votes are encrypted with the creator's public key
    19→    - Only the poll creator can see individual votes
    20→    - Results are shared as an aggregate summary
    21→    - Anonymous mode: even the creator cannot link votes to voters
    22→    - Secret ballot: votes are committed before reveal (hash commitment)
    23→
    24→Wire format for PollDefinition:
    25→    [1B version][2B question_len][N question]
    26→    [1B option_count][per option: 2B len + N text]
    27→    [1B settings_flags][4B close_timestamp][16B poll_id]
    28→
    29→Wire format for PollVote:
    30→    [1B version][16B poll_id][1B option_index][32B commitment_hash]
    31→    [8B timestamp][64B ed25519_signature]
    32→"""
    33→

    94→class PollType(enum.IntEnum):
    95→    """Type of poll."""
    96→    SINGLE_CHOICE = 0x01     # Select one option

   100→class PollVisibility(enum.IntEnum):
   101→    """Visibility of poll results."""
   102→    PUBLIC = 0x01            # Results visible as votes come in

   106→class PollSettings(enum.IntFlag):
   107→    """Bit flags for poll configuration."""
   108→    NONE = 0
   109→    ANONYMOUS = 0x01         # Voters are anonymous (even to creator)
   110→    SECRET_BALLOT = 0x02     # Commit-reveal scheme for votes
   111→    MULTIPLE_CHOICE = 0x04   # Allow selecting multiple options
   112→    ALLOW_CHANGE = 0x08      # Allow voters to change their vote

   116→    """Current status of a poll."""

   140→@dataclass
   141→class PollDefinition:
   142→    """Definition of a poll with question, options, and settings.
   143→
   144→    Attributes:
   145→        poll_id: 16-byte unique poll identifier.
   146→        question: The poll question text.
   147→        options: List of available options.
   148→        poll_type: Single or multiple choice.
   149→        visibility: Public or secret results.
   150→        settings: Bit flags for additional settings.
   151→        creator_user_id: User ID of the poll creator.
   152→        created_at: Unix timestamp of creation.
   153→        close_at: Unix timestamp when the poll auto-closes (0 = manual).
   154→        status: Current poll status.
   155→        total_voters: Total number of unique voters.
   156→    """
   157→    poll_id: bytes = b""
   158→    question: str = ""
   159→    options: list[PollOption] = field(default_factory=list)
   160→    poll_type: PollType = PollType.SINGLE_CHOICE
   161→    visibility: PollVisibility = PollVisibility.PUBLIC
   162→    settings: PollSettings = PollSettings.NONE
   163→    creator_user_id: str = ""
   164→    created_at: float = 0.0
   165→    close_at: float = 0.0
   166→    status: PollStatus = PollStatus.OPEN
   167→    total_voters: int = 0
   168→
   169→    def __post_init__(self) -> None:
   170→        if not self.poll_id:
   171→            self.poll_id = os.urandom(POLL_ID_SIZE)
   172→        if not self.created_at:
   173→            self.created_at = time.time()
   174→        if not self.options:
   175→            self.options = []
   176→
   177→    def encode(self) -> bytes:
   178→        """Serialize the poll definition for encrypted transport.
   179→
   180→        Returns:
   181→            Encoded bytes ready for E2EE encryption.
   182→        """
   183→        question_bytes = self.question.encode("utf-8")
   184→        if len(question_bytes) > MAX_QUESTION_LENGTH:
   185→            raise PollCreationError(
   186→                f"Question too long: {len(question_bytes)} bytes, max {MAX_QUESTION_LENGTH}."
   187→            )
   188→
   189→        # Header
   190→        settings_byte = (
   191→            (0x01 if PollSettings.ANONYMOUS in self.settings else 0)
   192→            | (0x02 if PollSettings.SECRET_BALLOT in self.settings else 0)
   193→            | (0x04 if PollSettings.MULTIPLE_CHOICE in self.settings else 0)
   194→            | (0x08 if PollSettings.ALLOW_CHANGE in self.settings else 0)
   195→        )
   196→
   197→        header = struct.pack(
   198→            "!BHBBIQ",
   199→            POLL_VERSION,
   200→            len(question_bytes),
   201→            len(self.options),
   202→            settings_byte,
   203→            int(self.close_at),
   204→            len(self.poll_id),
   205→        )
   206→
   207→        # Question
   208→        data = header + question_bytes
   209→
   210→        # Options
   211→        for opt in self.options:
   212→            opt_bytes = opt.text.encode("utf-8")
   213→            if len(opt_bytes) > MAX_OPTION_LENGTH:
   214→                raise PollCreationError(
   215→                    f"Option too long: {len(opt_bytes)} bytes, max {MAX_OPTION_LENGTH}."
   216→                )
   217→            data += struct.pack("!H", len(opt_bytes)) + opt_bytes
   218→
   219→        # Poll ID
   220→        data += self.poll_id
   221→
   222→        return data
   223→
   224→    @classmethod
   225→    def decode(cls, data: bytes) -> PollDefinition:
   226→        """Deserialize a poll definition from bytes.
   227→
   228→        Args:
   229→            data: Raw bytes from decrypted message.
   230→
   231→        Returns:
   232→            Parsed PollDefinition.
   233→
   234→        Raises:
   235→            PollError: If the data is malformed.
   236→        """
   237→        if len(data) < 16:
   238→            raise PollError(f"Poll data too short: {len(data)} bytes.")
   239→
   240→        offset = 0
   241→        version = data[offset]
   242→        offset += 1
   243→
   244→        if version != POLL_VERSION:
   245→            raise PollError(f"Unsupported poll version: {version}.")
   246→
   247→        question_len, option_count, settings_byte = struct.unpack(
   248→            "!HBB", data[offset:offset + 4]
   249→        )
   250→        offset += 4
   251→
   252→        close_at, poll_id_len = struct.unpack("!QI", data[offset:offset + 12])
   253→        offset += 12
   254→
   255→        # Question
   256→        if len(data) < offset + question_len:
   257→            raise PollError("Poll data truncated: missing question.")
   258→        question = data[offset:offset + question_len].decode("utf-8")
   259→        offset += question_len
   260→
   261→        # Options
   262→        options: list[PollOption] = []
   263→        for i in range(option_count):
   264→            if len(data) < offset + 2:
   265→                raise PollError("Poll data truncated: missing option length.")
   266→            opt_len = struct.unpack("!H", data[offset:offset + 2])[0]
   267→            offset += 2
   268→            if len(data) < offset + opt_len:
   269→                raise PollError("Poll data truncated: missing option text.")
   270→            opt_text = data[offset:offset + opt_len].decode("utf-8")
   271→            offset += opt_len
   272→            options.append(PollOption(text=opt_text, index=i))
   273→
   274→        # Poll ID
   275→        if len(data) < offset + poll_id_len:
   276→            raise PollError("Poll data truncated: missing poll ID.")
   277→        poll_id = data[offset:offset + poll_id_len]
   278→        offset += poll_id_len
   279→
   280→        # Parse settings
   281→        settings = PollSettings.NONE
   282→        if settings_byte & 0x01:
   283→            settings |= PollSettings.ANONYMOUS
   284→        if settings_byte & 0x02:
   285→            settings |= PollSettings.SECRET_BALLOT
   286→        if settings_byte & 0x04:
   287→            settings |= PollSettings.MULTIPLE_CHOICE
   288→        if settings_byte & 0x08:
   289→            settings |= PollSettings.ALLOW_CHANGE
   290→
   291→        poll_type = (
   292→            PollType.MULTIPLE_CHOICE
   293→            if PollSettings.MULTIPLE_CHOICE in settings
   294→            else PollType.SINGLE_CHOICE
   295→        )
   296→        visibility = (
   297→            PollVisibility.SECRET
   298→            if PollSettings.SECRET_BALLOT in settings
   299→            else PollVisibility.PUBLIC
   300→        )
   301→
   302→        return cls(
   303→            poll_id=poll_id,
   304→            question=question,
   305→            options=options,
   306→            poll_type=poll_type,
   307→            visibility=visibility,
   308→            settings=settings,
   309→            close_at=close_at,
   310→        )
   311→
   312→    def is_closed(self) -> bool:
   313→        """Check if the poll is closed (manually or by timeout)."""
   314→        if self.status != PollStatus.OPEN:
   315→            return True
   316→        if self.close_at > 0 and time.time() > self.close_at:
   317→            return True
   318→        return False
   319→
   320→    def get_winning_option(self) -> Optional[PollOption]:
   321→        """Get the option with the most votes.
   322→
   323→        Returns:
   324→            The winning PollOption, or None if tied or no votes.
   325→        """
   326→        if not self.options or self.total_voters == 0:
   327→            return None
   328→        max_votes = max(opt.vote_count for opt in self.options)
   329→        winners = [opt for opt in self.options if opt.vote_count == max_votes]
   330→        if len(winners) == 1:
   331→            return winners[0]
   332→        return None  # Tie
   333→
   334→
   335→@dataclass
   336→class PollVote:
   337→    """A single vote cast on a poll.
   338→
   339→    Attributes:
   340→        poll_id: The poll this vote is for.
   341→        option_index: Index of the selected option.
   342→        voter_user_id: User ID of the voter (empty if anonymous).
   343→        commitment_hash: Hash commitment for secret ballot.
   344→        timestamp: Unix timestamp when the vote was cast.
   345→        signature: Ed25519 signature over the vote data.
   346→    """
   347→    poll_id: bytes = b""
   348→    option_index: int = 0
   349→    voter_user_id: str = ""
   350→    commitment_hash: bytes = b""   # 32 bytes SHA-256
   351→    timestamp: float = 0.0
   352→    signature: bytes = b""         # 64 bytes Ed25519
   353→
   354→    def compute_commitment(self, nonce: bytes = b"") -> bytes:
   355→        """Compute a hash commitment for the vote.
   356→
   357→        Used in secret ballot mode to allow voters to commit to a vote
   358→        without revealing their choice until the reveal phase.
   359→
   360→        Args:
   361→            nonce: Random nonce for the commitment (auto-generated if empty).
   362→
   363→        Returns:
   364→            32-byte SHA-256 commitment hash.
   365→        """
   366→        if not nonce:
   367→            nonce = os.urandom(16)
   368→        data = (
   369→            self.poll_id
   370→            + struct.pack("!B", self.option_index)
   371→            + nonce
   372→            + self.voter_user_id.encode("utf-8")
   373→        )
   374→        return hashlib.sha256(data).digest()
   375→
   376→    def encode(self) -> bytes:
   377→        """Serialize the vote for encrypted transport.
   378→
   379→        Returns:
   380→            Encoded bytes ready for E2EE encryption.
   381→        """
   382→        voter_bytes = self.voter_user_id.encode("utf-8")
   383→
   384→        data = struct.pack(
   385→            "!B16sB32sQH",
   386→            POLL_VERSION,
   387→            self.poll_id[:POLL_ID_SIZE],
   388→            self.option_index,
   389→            self.commitment_hash,
   390→            int(self.timestamp),
   391→            len(voter_bytes),
   392→        )
   393→        data += voter_bytes
   394→        data += self.signature[:64]
   395→
   396→        return data
   397→
   398→    @classmethod
   399→    def decode(cls, data: bytes) -> PollVote:
   400→        """Deserialize a vote from bytes.
   401→
   402→        Args:
   403→            data: Raw bytes from decrypted message.
   404→
   405→        Returns:
   406→            Parsed PollVote.
   407→
   408→        Raises:
   409→            VoteError: If the data is malformed.
   410→        """
   411→        min_size = 1 + 16 + 1 + 32 + 8 + 2
   412→        if len(data) < min_size:
   413→            raise VoteError(f"Vote data too short: {len(data)} bytes.")
   414→
   415→        offset = 0
   416→        version = data[offset]
   417→        offset += 1
   418→
   419→        if version != POLL_VERSION:
   420→            raise VoteError(f"Unsupported vote version: {version}.")
   421→
   422→        poll_id = data[offset:offset + POLL_ID_SIZE]
   423→        offset += POLL_ID_SIZE
   424→
   425→        option_index = data[offset]
   426→        offset += 1
   427→
   428→        commitment = data[offset:offset + 32]
   429→        offset += 32
   430→
   431→        timestamp, voter_len = struct.unpack("!QH", data[offset:offset + 10])
   432→        offset += 10
   433→
   434→        if len(data) < offset + voter_len + 64:
   435→            raise VoteError("Vote data truncated.")
   436→
   437→        voter_user_id = data[offset:offset + voter_len].decode("utf-8")
   438→        offset += voter_len
   439→
   440→        signature = data[offset:offset + 64]
   441→
   442→        return cls(
   443→            poll_id=poll_id,
   444→            option_index=option_index,
   445→            voter_user_id=voter_user_id,
   446→            commitment_hash=commitment,
   447→            timestamp=timestamp,
   448→            signature=signature,
   449→        )
   450→

   452→@dataclass
   453→class PollResults:
   454→    """Aggregated results for a closed poll.
   455→
   456→    Attributes:
   457→        poll_id: The poll these results are for.
   458→        option_results: Mapping of option_index to vote count.
   459→        total_voters: Total number of unique voters.
   460→        winning_option: Index of the winning option (-1 if tied).
   461→        closed_at: Unix timestamp when the poll was closed.
   462→    """
   463→    poll_id: bytes = b""
   464→    option_results: dict[int, int] = field(default_factory=dict)
   465→    total_voters: int = 0
   466→    winning_option: int = -1
   467→    closed_at: float = 0.0
   468→
   469→    def encode(self) -> bytes:
   470→        """Serialize poll results for broadcast."""
   471→        data = struct.pack("!BIQ", POLL_VERSION, self.total_voters, int(self.closed_at))
   472→        data += self.poll_id[:POLL_ID_SIZE]
   473→        data += struct.pack("!H", len(self.option_results))
   474→        for opt_idx, count in self.option_results.items():
   475→            data += struct.pack("!BI", opt_idx, count)
   476→        data += struct.pack("!b", self.winning_option)
   477→        return data
   478→
   479→    @classmethod
   480→    def decode(cls, data: bytes) -> PollResults:
   481→        """Deserialize poll results."""
   482→        if len(data) < 15:
   483→            raise PollError("Results data too short.")
   484→
   485→        offset = 0
   486→        version = data[offset]
   487→        offset += 1
   488→
   489→        total_voters, closed_at = struct.unpack("!IQ", data[offset:offset + 12])
   490→        offset += 12
   491→
   492→        poll_id = data[offset:offset + POLL_ID_SIZE]
   493→        offset += POLL_ID_SIZE
   494→
   495→        result_count = struct.unpack("!H", data[offset:offset + 2])[0]
   496→        offset += 2
   497→
   498→        option_results: dict[int, int] = {}
   499→        for _ in range(result_count):
   500→            opt_idx, count = struct.unpack("!BI", data[offset:offset + 5])
   501→            offset += 5
   502→            option_results[opt_idx] = count
   503→
   504→        winning = struct.unpack("!b", data[offset:offset + 1])[0]
   505→
   506→        return cls(
   507→            poll_id=poll_id,
   508→            option_results=option_results,
   509→            total_voters=total_voters,
   510→            winning_option=winning,
   511→            closed_at=float(closed_at),
   512→        )
   513→

   515→# ---------------------------------------------------------------------------
   516→# Poll Manager
   517→# ---------------------------------------------------------------------------
   518→
   519→class PollManager:
   520→    """Manages poll lifecycle: creation, voting, tallying, and results.
   521→
   522→    The PollManager handles both creator-side operations (creating polls,
   523→    receiving and tallying votes, broadcasting results) and voter-side
   524→    operations (casting votes, receiving results).
   525→
   526→    For the creator:
   527→        - create_poll(): Create and encode a new poll
   528→        - receive_vote(): Process an incoming encrypted vote
   529→        - close_poll(): Close a poll and broadcast results
   530→
   531→    For the voter:
   532→        - cast_vote(): Create and encode a vote
   533→        - receive_results(): Process incoming poll results
   534→
   535→    Lifecycle:
   536→        manager = PollManager(user_id="abc123", keypair=ed25519)
   537→        # Creator
   538→        poll = manager.create_poll("What time?", ["9am", "10am", "11am"])
   539→        # Voter
   540→        vote = manager.cast_vote(poll.poll_id, option_index=1)
   541→        # Creator
   542→        manager.receive_vote(vote)
   543→        results = manager.close_poll(poll.poll_id)
   544→    """
   545→
   546→    def __init__(
   547→        self,
   548→        user_id: str = "",
   549→        keypair: Optional[object] = None,
   550→    ) -> None:
   551→        self._user_id = user_id
   552→        self._keypair = keypair
   553→        self._active_polls: dict[str, PollDefinition] = {}
   554→        self._votes: dict[str, list[PollVote]] = {}      # poll_id → votes
   555→        self._voted_by: dict[str, set[str]] = {}          # poll_id → set of voter_ids
   556→        self._results: dict[str, PollResults] = {}
   557→
   558→        # Callbacks
   559→        self._on_poll_created: Optional[Callable[[PollDefinition], Awaitable[None]]] = None
   560→        self._on_vote_received: Optional[Callable[[PollVote], Awaitable[None]]] = None
   561→        self._on_results_received: Optional[Callable[[PollResults], Awaitable[None]]] = None
   562→
   563→    # -------------------------------------------------------------------
   564→    # Callback registration
   565→    # -------------------------------------------------------------------
   566→
   567→    def set_on_poll_created(
   568→        self, callback: Callable[[PollDefinition], Awaitable[None]]
   569→    ) -> None:
   570→        self._on_poll_created = callback
   571→
   572→    def set_on_vote_received(
   573→        self, callback: Callable[[PollVote], Awaitable[None]]
   574→    ) -> None:
   575→        self._on_vote_received = callback
   576→
   577→    def set_on_results_received(
   578→        self, callback: Callable[[PollResults], Awaitable[None]]
   579→    ) -> None:
   580→        self._on_results_received = callback
   581→
   582→    # -------------------------------------------------------------------
   583→    # Creator operations
   584→    # -------------------------------------------------------------------
   585→
   586→    def create_poll(
   587→        self,
   588→        question: str,
   589→        options: list[str],
   590→        poll_type: PollType = PollType.SINGLE_CHOICE,
   591→        visibility: PollVisibility = PollVisibility.PUBLIC,
   592→        settings: PollSettings = PollSettings.NONE,
   593→        close_after_hours: float = 24.0,
   594→    ) -> PollDefinition:
   595→        """Create a new poll.
   596→
   597→        Args:
   598→            question: The poll question.
   599→            options: List of option texts (2-20 options).
   600→            poll_type: Single or multiple choice.
   601→            visibility: Public or secret results.
   602→            settings: Additional settings flags.
   603→            close_after_hours: Hours until auto-close (0 = manual).
   604→
   605→        Returns:
   606→            The created PollDefinition.
   607→
   608→        Raises:
   609→            PollCreationError: If the poll parameters are invalid.
   610→        """
   611→        if not question or len(question) > MAX_QUESTION_LENGTH:
   612→            raise PollCreationError(
   613→                f"Question must be 1-{MAX_QUESTION_LENGTH} characters."
   614→            )
   615→        if len(options) < MIN_OPTIONS:
   616→            raise PollCreationError(f"Need at least {MIN_OPTIONS} options.")
   617→        if len(options) > MAX_OPTIONS:
   618→            raise PollCreationError(f"Maximum {MAX_OPTIONS} options allowed.")
   619→        if close_after_hours > MAX_POLL_DURATION_HOURS:
   620→            raise PollCreationError(
   621→                f"Duration cannot exceed {MAX_POLL_DURATION_HOURS} hours."
   622→            )
   623→
   624→        poll_id = os.urandom(POLL_ID_SIZE)
   625→        poll_options = [
   626→            PollOption(text=text, index=i)
   627→            for i, text in enumerate(options)
   628→        ]
   629→
   630→        close_at = (
   631→            time.time() + (close_after_hours * 3600)
   632→            if close_after_hours > 0
   633→            else 0
   634→        )
   635→
   636→        poll = PollDefinition(
   637→            poll_id=poll_id,
   638→            question=question,
   639→            options=poll_options,
   640→            poll_type=poll_type,
   641→            visibility=visibility,
   642→            settings=settings,
   643→            creator_user_id=self._user_id,
   644→            close_at=close_at,
   645→        )
   646→
   647→        # Register the poll
   648→        poll_key = poll_id.hex()
   649→        self._active_polls[poll_key] = poll
   650→        self._votes[poll_key] = []
   651→        self._voted_by[poll_key] = set()
   652→
   653→        logger.info(
   654→            "Created poll '%s' (%d options, type=%s, visibility=%s).",
   655→            question[:50],
   656→            len(options),
   657→            poll_type.name,
   658→            visibility.name,
   659→        )
   660→
   661→        if self._on_poll_created:
   662→            asyncio.ensure_future(self._on_poll_created(poll))
   663→
   664→        return poll
   665→
   666→    def receive_vote(self, vote: PollVote) -> bool:
   667→        """Process an incoming vote on one of our polls.
   668→
   669→        Args:
   670→            vote: The received PollVote.
   671→
   672→        Returns:
   673→            True if the vote was accepted, False if rejected.
   674→
   675→        Raises:
   676→            PollClosedError: If the poll is closed.
   677→            DuplicateVoteError: If the voter has already voted.
   678→        """
   679→        poll_key = vote.poll_id.hex()
   680→        poll = self._active_polls.get(poll_key)
   681→
   682→        if poll is None:
   683→            raise PollNotFoundError(f"Poll {poll_key[:16]} not found.")
   684→
   685→        if poll.is_closed():
   686→            raise PollClosedError(f"Poll '{poll.question}' is closed.")
   687→
   688→        # Check for duplicate vote
   689→        voter_id = vote.voter_user_id
   690→        if voter_id in self._voted_by.get(poll_key, set()):
   691→            if PollSettings.ALLOW_CHANGE not in poll.settings:
   692→                raise DuplicateVoteError(
   693→                    f"User {voter_id[:8]} already voted on this poll."
   694→                )
   695→            # Remove previous vote if change is allowed
   696→            self._votes[poll_key] = [
   697→                v for v in self._votes[poll_key]
   698→                if v.voter_user_id != voter_id
   699→            ]
   700→            logger.debug("Vote change allowed for %s.", voter_id[:8])
   701→
   702→        # Validate option index
   703→        if vote.option_index < 0 or vote.option_index >= len(poll.options):
   704→            raise VoteError(
   705→                f"Invalid option index: {vote.option_index}, "
   706→                f"poll has {len(poll.options)} options."
   707→            )
   708→
   709→        # Record the vote
   710→        self._votes[poll_key].append(vote)
   711→        self._voted_by.setdefault(poll_key, set()).add(voter_id)
   712→
   713→        # Update vote count
   714→        poll.options[vote.option_index].vote_count += 1
   715→        poll.total_voters = len(self._voted_by[poll_key])
   716→
   717→        logger.debug(
   718→            "Vote received on poll '%s': option %d by %s",
   719→            poll.question[:30],
   720→            vote.option_index,
   721→            voter_id[:8] if voter_id else "anonymous",
   722→        )
   723→
   724→        if self._on_vote_received:
   725→            asyncio.ensure_future(self._on_vote_received(vote))
   726→
   727→        return True
   728→
   729→    def close_poll(self, poll_id: bytes) -> PollResults:
   730→        """Close a poll and compute final results.
   731→
   732→        Args:
   733→            poll_id: The 16-byte poll identifier.
   734→
   735→        Returns:
   736→            PollResults with the final tallied results.
   737→
   738→        Raises:
   739→            PollNotFoundError: If the poll doesn't exist.
   740→        """
   741→        poll_key = poll_id.hex()
   742→        poll = self._active_polls.get(poll_key)
   743→
   744→        if poll is None:
   745→            raise PollNotFoundError(f"Poll {poll_key[:16]} not found.")
   746→
   747→        poll.status = PollStatus.CLOSED
   748→
   749→        # Compute results
   750→        option_results = {
   751→            opt.index: opt.vote_count for opt in poll.options
   752→        }
   753→        winning = poll.get_winning_option()
   754→        winning_idx = winning.index if winning else -1
   755→
   756→        results = PollResults(
   757→            poll_id=poll_id,
   758→            option_results=option_results,
   759→            total_voters=poll.total_voters,
   760→            winning_option=winning_idx,
   761→            closed_at=time.time(),
   762→        )
   763→
   764→        self._results[poll_key] = results
   765→
   766→        logger.info(
   767→            "Poll '%s' closed: %d voters, winner=%s",
   768→            poll.question[:30],
   769→            poll.total_voters,
   770→            winning.text if winning else "tie",
   771→        )
   772→
   773→        if self._on_results_received:
   774→            asyncio.ensure_future(self._on_results_received(results))
   775→
   776→        return results

   782→    def cast_vote(
   783→        self,
   784→        poll_id: bytes,
   785→        option_index: int,
   786→        voter_user_id: str = "",
   787→    ) -> PollVote:
   788→        """Create a vote for a poll.
   789→
   790→        Args:
   791→            poll_id: The 16-byte poll identifier.
   792→            option_index: Index of the selected option.
   793→            voter_user_id: Our user ID (empty for anonymous).
   794→
   795→        Returns:
   796→            The created PollVote, ready for encryption and sending.
   797→
   798→        Raises:
   799→            VoteError: If the vote parameters are invalid.
   800→        """
   801→        poll_key = poll_id.hex()
   802→        poll = self._active_polls.get(poll_key)
   803→
   804→        if poll and poll.is_closed():
   805→            raise PollClosedError("Cannot vote on a closed poll.")
   806→
   807→        if poll and (option_index < 0 or option_index >= len(poll.options)):
   808→            raise VoteError(
   809→                f"Invalid option index: {option_index}."
   810→            )
   811→
   812→        # Determine voter identity
   813→        if poll and PollSettings.ANONYMOUS in poll.settings:
   814→            actual_voter_id = ""  # Anonymous
   815→        else:
   816→            actual_voter_id = voter_user_id or self._user_id
   817→
   818→        vote = PollVote(
   819→            poll_id=poll_id,
   820→            option_index=option_index,
   821→            voter_user_id=actual_voter_id,
   822→            timestamp=time.time(),
   823→        )
   824→
   825→        # Compute commitment for secret ballot
   826→        if poll and PollSettings.SECRET_BALLOT in poll.settings:
   827→            vote.commitment_hash = vote.compute_commitment()
   828→        else:
   829→            vote.commitment_hash = b"\x00" * 32
   830→
   831→        # Sign the vote if we have a keypair
   832→        if self._keypair is not None:
   833→            try:
   834→                vote.signature = self._keypair.sign(vote.encode())
   835→            except Exception:
   836→                vote.signature = b"\x00" * 64
   837→
   838→        logger.debug(
   839→            "Vote cast on poll %s: option %d",
   840→            poll_key[:8],
   841→            option_index,
   842→        )
   843→
   844→        return vote

   846→    def receive_results(self, results: PollResults) -> None:
   847→        """Process received poll results.
   848→
   849→        Updates the local poll definition with the final tallied results.
   850→
   851→        Args:
   852→            results: The received PollResults.
   853→        """
   854→        poll_key = results.poll_id.hex()
   855→        poll = self._active_polls.get(poll_key)
   856→
   857→        if poll is None:
   858→            logger.warning("Received results for unknown poll %s.", poll_key[:8])
   859→            return
   860→
   861→        poll.status = PollStatus.CLOSED
   862→        poll.total_voters = results.total_voters
   863→
   864→        # Update option counts from results
   865→        for opt_idx, count in results.option_results.items():
   866→            if 0 <= opt_idx < len(poll.options):
   867→                poll.options[opt_idx].vote_count = count
   868→
   869→        self._results[poll_key] = results
   870→
   871→        if self._on_results_received:
   872→            asyncio.ensure_future(self._on_results_received(results))
   873→
   874→        logger.info(
   875→            "Poll results received for '%s': %d voters.",
   876→            poll.question[:30],
   877→            results.total_voters,
   878→        )
