   360→        """Full discovery protocol: register → discover → match."""
   361→        from presidium.core.network.discovery_v2 import ContactDiscoveryV2
   362→
   363→        # Alice registers her contacts
   364→        alice = ContactDiscoveryV2(user_id="alice", key_bits=1024)
   365→        await alice.initialize()
   366→        await alice.register_contacts(["+15551234567", "bob@example.com"])
   367→
   368→        # Bob also registers "bob@example.com"

   371→        await bob.initialize()
   372→        await bob.register_contacts(["bob@example.com", "carol@example.com"])
   373→
   374→        # Alice discovers using Bob's server
   375→        # (In production the server is shared; here we use Alice's server

   388→        await alice.shutdown()
   389→        await bob.shutdown()

   466→    def test_blob_expiry(self) -> None:
   467→        """Blobs should be evicted after their expiry time."""
   468→        from presidium.push.relay_server import PushRelayServer
   469→        server = PushRelayServer()
   470→        server.start()
   471→
   472→        blob_id = hashlib.sha256(b"test-expiry").digest()[:16]
   473→        encrypted_blob = os.urandom(128)
   474→        hmac_value = hashlib.sha256(blob_id + encrypted_blob).digest()
   475→
   476→        # Store with immediate expiry (1 second)
   477→        server.store_blob(blob_id, encrypted_blob, hmac_value, ttl_seconds=1)
   478→
   479→        # Should be available now
   480→        assert server.get_blob(blob_id) == encrypted_blob
   481→
   482→        # Wait for expiry
   483→        time.sleep(1.5)

   485→
   486→        # Should be gone
   487→        assert server.get_blob(blob_id) is None
   488→        server.stop()
