     1→     1→"""Push Client for Presidium — privacy-preserving notification receiver.
     2→     2→
     3→     3→Implements the client-side of the Zero-Knowledge Push system:
     4→     4→    - Blind fetch: computes expected blob IDs from own public key and
     5→     5→      timestamp buckets, then fetches matching blobs without revealing
     6→     6→      identity to the relay
     7→     7→    - FCM/APNS fallback: receives only a 16-byte encrypted_payload_id,
     8→     8→      then performs a blind fetch to retrieve the actual E2EE payload
     9→     9→    - Local decryption: uses Double Ratchet / HPKE to decrypt payloads
    10→    10→      that the relay could never read
    11→    11→    - Background polling: periodic fetch on configurable interval
    12→    12→
    13→    13→Protocol flow:
    14→    14→    Client (background)                     Relay
    15→    15→      |                                       |
    16→    16→      |  1. Compute blob_ids:                 |
    17→    17→      |     id = SHA-256(my_pk || bucket)     |
    18→    18→      |     bucket = floor(now / 3600)        |
    19→    19→      |  2. GET /pull?ids=<hex_id1>,...       |
    20→    20→      |  ----ids (no identity)------------>   |
    21→    21→      |  <---encrypted blobs----------------  |
    22→    22→      |  3. Decrypt with Double Ratchet/HPKE  |
    23→    23→      |                                       |
    24→    24→      |                                       |
    25→    25→    FCM/APNS path:                           |
    26→    26→      |  <---push with blob_id only--------   |
    27→    27→      |  GET /pull?ids=<hex_blob_id>          |
    28→    28→      |  ----blob_id---------------------->   |
    29→    29→      |  <---encrypted blob-----------------  |
    30→    30→      |  4. Decrypt locally                  |
    31→    31→
    32→    32→Privacy guarantees:
    33→    33→    - Relay never learns which public key maps to which blob IDs
    34→    34→    - Relay never sees plaintext message content
    35→    35→    - FCM/APNS only receives 16-byte blob_id, no message metadata
    36→    36→    - Blob IDs change hourly, preventing long-term tracking
    37→    37→    - Each blob is fetched only once (forward secrecy)
    38→    38→"""

   170→   170→# ---------------------------------------------------------------------------
   171→   171→# Blob ID computation
   172→   172→# ---------------------------------------------------------------------------
   173→   173→
   174→   174→class BlobIDComputer:
   175→   175→    """Computes pseudorandom blob IDs from a public key and time buckets.
   176→   176→
   177→   177→    The recipient precomputes the blob IDs they expect to have messages
   178→   178→    for, then fetches them in a single blind request. The relay cannot
   179→   179→    determine which public key corresponds to which blob IDs because
   180→   180→    SHA-256 is a one-way function.
   181→   181→
   182→   182→    Formula:
   183→   183→        bucket = floor(unix_timestamp / TIME_BUCKET_SECONDS)
   184→   184→        blob_id = SHA-256(public_key || big_endian_uint64(bucket))[:16]
   185→   185→    """
   186→   186→
   187→   187→    def __init__(self, public_key: bytes) -> None:
   188→   188→        if len(public_key) != 32:
   189→   189→            raise PushClientError(
   190→   190→                f"Invalid public key size: {len(public_key)}, expected 32."
   191→   191→            )
   192→   192→        self._public_key = public_key
   193→   193→
   194→   194→    def compute_current_ids(self, num_buckets: int = 3) -> list[bytes]:
   195→   195→        """Compute blob IDs for the current and recent time buckets.
   196→   196→
   197→   197→        Args:
   198→   198→            num_buckets: Number of buckets to compute (current + past).
   199→   199→
   200→   200→        Returns:
   201→   201→            List of 16-byte blob IDs, most recent first.
   202→   202→        """
   203→   203→        current_bucket = int(time.time()) // TIME_BUCKET_SECONDS
   204→   204→        ids: list[bytes] = []
   205→   205→        for offset in range(num_buckets):
   206→   206→            bucket = current_bucket - offset
   207→   207→            blob_id = self._compute_id(bucket)
   208→   208→            ids.append(blob_id)
   209→   209→        return ids
   210→   210→
   211→   211→    def compute_id_for_blob_id_hex(self, blob_id_hex: str) -> bytes:
   212→   212→        """Convert a hex blob ID from an FCM/APNS push to bytes.
   213→   213→
   214→   214→        Args:
   215→   215→            blob_id_hex: 32-character hex string of the blob ID.
   216→   216→
   217→   217→        Returns:
   218→   218→            16-byte blob ID.
   219→   219→        """
   220→   220→        try:
   221→   221→            bid = bytes.fromhex(blob_id_hex)
   222→   222→            if len(bid) != BLOB_ID_SIZE:
   223→   223→                raise PushClientError(
   224→   224→                    f"Invalid blob ID size: {len(bid)}, expected {BLOB_ID_SIZE}."
   225→   225→                )
   226→   226→            return bid
   227→   227→        except ValueError as exc:
   228→   228→            raise PushClientError(f"Invalid blob ID hex: {exc}") from exc
   229→   229→
   230→   230→    def _compute_id(self, bucket: int) -> bytes:
   231→   231→        """Compute a single blob ID for a time bucket.
   232→   232→
   233→   233→        Args:
   234→   234→            bucket: Time bucket index.
   235→   235→
   236→   236→        Returns:
   237→   237→            16-byte blob ID.
   238→   238→        """
   239→   239→        data = self._public_key + struct.pack("!Q", bucket)
   240→   240→        return hashlib.sha256(data).digest()[:BLOB_ID_SIZE]
   241→   241→

   246→   246→
   247→   247→class PushClient:
   248→   248→    """Client for the Zero-Knowledge Push Relay.
   249→   249→
   250→   250→    Provides:
   251→   251→        - Background polling for new encrypted blobs
   252→   252→        - FCM/APNS-triggered fetch on receiving a push
   253→   253→        - Device token registration with the relay
   254→   254→        - Local decryption of relay-level encryption
   255→   255→        - Callbacks for delivering decrypted notifications to the app
   256→   256→
   257→   257→    Lifecycle:
   258→   258→        client = PushClient(config)
   259→   259→        client.set_on_notification(callback)
   260→   260→        await client.start()    # Starts background polling
   261→   261→        ...
   262→   262→        await client.stop()     # Graceful shutdown
   263→   263→    """
   264→   264→
   265→   265→    def __init__(self, config: PushClientConfig) -> None:
   266→   266→        self._config = config
   267→   267→        self._id_computer = BlobIDComputer(config.public_key)
   268→   268→        self._session: Optional[aiohttp.ClientSession] = None
   269→   269→        self._poll_task: Optional[asyncio.Task] = None
   270→   270→        self._running = False
   271→   271→        self._last_poll_time: float = 0.0
   272→   272→        self._consecutive_errors = 0
   273→   273→
   274→   274→        # Callbacks
   275→   275→        self._on_notification: Optional[Callable[[DecryptedNotification], Awaitable[None]]] = None
   276→   276→        self._on_fetch_error: Optional[Callable[[Exception], Awaitable[None]]] = None
   277→   277→
   278→   278→        # Stats
   279→   279→        self._total_fetched: int = 0
   280→   280→        self._total_decrypted: int = 0
   281→   281→        self._total_errors: int = 0
   282→   282→

   290→   290→        """Register a callback for when a notification is decrypted.

   311→   311→    async def start(self) -> None:
   312→   312→        """Start the push client background polling.
   313→   313→
   314→   314→        Creates an aiohttp session and starts the poll loop. If a
   315→   315→        device token is configured, also registers with the relay.
   316→   316→        """
   317→   317→        if self._running:
   318→   318→            return
   319→   319→
   320→   320→        self._session = aiohttp.ClientSession(
   321→   321→            timeout=aiohttp.ClientTimeout(total=self._config.pull_timeout),
   322→   322→        )
   323→   323→        self._running = True
   324→   324→
   325→   325→        # Register device token if available
   326→   326→        if self._config.provider in (PushProvider.FCM, PushProvider.APNS):
   327→   327→            await self._register_device_token()
   328→   328→
   329→   329→        # Start background polling
   330→   330→        self._poll_task = asyncio.create_task(self._poll_loop())
   331→   331→
   332→   332→        logger.info(
   333→   333→            "Push client started (provider=%s, interval=%ds).",
   334→   334→            self._config.provider.value,
   335→   335→            self._config.poll_interval,
   336→   336→        )
   337→   337→
   338→   338→    async def stop(self) -> None:
   339→   339→        """Stop the push client and clean up resources."""
   340→   340→        self._running = False
   341→   341→
   342→   342→        if self._poll_task is not None:
   343→   343→            self._poll_task.cancel()
   344→   344→            try:
   345→   345→                await self._poll_task
   346→   346→            except asyncio.CancelledError:
   347→   347→                pass
   348→   348→            self._poll_task = None
   349→   349→
   350→   350→        if self._session is not None:
   351→   351→            await self._session.close()
   352→   352→            self._session = None
   353→   353→
   354→   354→        logger.info("Push client stopped.")
   355→   355→
   356→   356→    # -------------------------------------------------------------------
   357→   357→    # Background polling
   358→   358→    # -------------------------------------------------------------------
   359→   359→
   360→   360→    async def _poll_loop(self) -> None:
   361→   361→        """Periodically poll the relay for new blobs."""
   362→   362→        while self._running:
   363→   363→            try:
   364→   364→                await asyncio.sleep(self._config.poll_interval)
   365→   365→                if not self._running:
   366→   366→                    break
   367→   367→
   368→   368→                await self.fetch_and_decrypt()
   369→   369→                self._consecutive_errors = 0
   370→   370→
   371→   371→            except asyncio.CancelledError:
   372→   372→                raise
   373→   373→            except Exception as exc:
   374→   374→                self._consecutive_errors += 1
   375→   375→                self._total_errors += 1
   376→   376→                logger.warning(
   377→   377→                    "Poll error (#%d): %s", self._consecutive_errors, exc
   378→   378→                )
   379→   379→
   380→   380→                if self._on_fetch_error:
   381→   381→                    await self._on_fetch_error(exc)
   382→   382→
   383→   383→                # Exponential backoff on repeated errors
   384→   384→                if self._consecutive_errors > 1:
   385→   385→                    backoff = min(
   386→   386→                        self._config.poll_interval * (2 ** min(self._consecutive_errors - 1, 5)),
   387→   387→                        300,  # Max 5 minutes
   388→   388→                    )
   389→   389→                    await asyncio.sleep(backoff)
   390→   390→
   391→   391→    # -------------------------------------------------------------------
   392→   392→    # Fetch and decrypt
   393→   393→    # -------------------------------------------------------------------
   394→   394→
   395→   395→    async def fetch_and_decrypt(self) -> list[DecryptedNotification]:
   396→   396→        """Fetch new blobs from the relay and decrypt them.
   397→   397→
   398→   398→        Computes expected blob IDs from our public key and current/recent
   399→   399→        time buckets, then fetches matching blobs and decrypts them.
   400→   400→
   401→   401→        Returns:
   402→   402→            List of decrypted notifications.
   403→   403→        """
   404→   404→        if self._session is None:
   405→   405→            raise FetchError("Client not started — no HTTP session.")
   406→   406→
   407→   407→        # Compute blob IDs
   408→   408→        blob_ids = self._id_computer.compute_current_ids(
   409→   409→            num_buckets=self._config.time_buckets
   410→   410→        )
   411→   411→
   412→   412→        # Fetch from relay
   413→   413→        blobs = await self._fetch_blobs(blob_ids)
   414→   414→        self._total_fetched += len(blobs)
   415→   415→
   416→   416→        # Decrypt each blob
   417→   417→        notifications: list[DecryptedNotification] = []
   418→   418→        for blob in blobs:
   419→   419→            try:
   420→   420→                notification = self._decrypt_blob(blob)
   421→   421→                notifications.append(notification)
   422→   422→                self._total_decrypted += 1
   423→   423→
   424→   424→                # Deliver via callback
   425→   425→                if self._on_notification:
   426→   426→                    await self._on_notification(notification)
   427→   427→
   428→   428→            except Exception as exc:
   429→   429→                logger.warning(
   430→   430→                    "Failed to decrypt blob %s: %s",
   431→   431→                    blob.blob_id.hex(),
   432→   432→                    exc,
   433→   433→                )
   434→   434→                self._total_errors += 1
   435→   435→
   436→   436→        self._last_poll_time = time.time()
   437→   437→        return notifications
   438→   438→
   439→   439→    async def fetch_by_blob_id(self, blob_id_hex: str) -> Optional[DecryptedNotification]:
   440→   440→        """Fetch a specific blob by its hex ID (from FCM/APNS push).
   441→   441→
   442→   442→        This is called when the app receives a push notification
   443→   443→        containing only a blob_id, and needs to fetch the actual
   444→   444→        encrypted payload.
   445→   445→
   446→   446→        Args:
   447→   447→            blob_id_hex: 32-character hex string of the blob ID.
   448→   448→
   449→   449→        Returns:
   450→   450→            Decrypted notification, or None if blob not found.
   451→   451→        """
   452→   452→        if self._session is None:
   453→   453→            raise FetchError("Client not started — no HTTP session.")
   454→   454→
   455→   455→        blob_id = self._id_computer.compute_id_for_blob_id_hex(blob_id_hex)
   456→   456→        blobs = await self._fetch_blobs([blob_id])
   457→   457→
   458→   458→        if not blobs:
   459→   459→            logger.debug("Blob %s not found on relay.", blob_id_hex)
   460→   460→            return None
   461→   461→
   462→   462→        notification = self._decrypt_blob(blobs[0])
   463→   463→        self._total_fetched += 1
   464→   464→        self._total_decrypted += 1
   465→   465→
   466→   466→        if self._on_notification:
   467→   467→            await self._on_notification(notification)
   468→   468→
   469→   469→        return notification
   470→   470→

   472→   472→    # HTTP operations
   473→   473→    # -------------------------------------------------------------------
   474→   474→
   475→   475→    async def _fetch_blobs(self, blob_ids: list[bytes]) -> list[FetchedBlob]:
   476→   476→        """Fetch blobs from the relay by their IDs.
   477→   477→
   478→   478→        Args:
   479→   479→            blob_ids: List of 16-byte blob identifiers.
   480→   480→
   481→   481→        Returns:
   482→   482→            List of FetchedBlob objects.
   483→   483→
   484→   484→        Raises:
   485→   485→            FetchError: If the HTTP request fails.
   486→   486→            RelayUnavailableError: If the relay is unreachable.
   487→   487→        """
   488→   488→        ids_param = ",".join(bid.hex() for bid in blob_ids)
   489→   489→        url = f"{self._config.relay_url}/pull?ids={ids_param}"
   490→   490→
   491→   491→        for attempt in range(MAX_POLL_RETRIES):
   492→   492→            try:
   493→   493→                async with self._session.get(url) as resp:
   494→   494→                    if resp.status == 200:
   495→   495→                        data = await resp.json()
   496→   496→                        return self._parse_pull_response(data)
   497→   497→                    if resp.status == 429:
   498→   498→                        retry_after = int(resp.headers.get("Retry-After", "60"))
   499→   499→                        logger.warning("Rate limited. Retrying in %ds.", retry_after)
   500→   500→                        await asyncio.sleep(retry_after)
   501→   501→                        continue
   502→   502→                    if resp.status >= 500:
   503→   503→                        logger.warning("Relay error %d, retrying...", resp.status)
   504→   504→                        await asyncio.sleep(2 ** attempt)
   505→   505→                        continue
   506→   506→                    # Client error — don't retry
   507→   507→                    body = await resp.text()
   508→   508→                    raise FetchError(f"Relay returned {resp.status}: {body[:200]}")
   509→   509→
   510→   510→            except aiohttp.ClientError as exc:
   511→   511→                if attempt == MAX_POLL_RETRIES - 1:
   512→   512→                    raise RelayUnavailableError(
   513→   513→                        f"Relay unreachable after {MAX_POLL_RETRIES} attempts: {exc}"
   514→   514→                    ) from exc
   515→   515→                await asyncio.sleep(2 ** attempt)
   516→   516→
   517→   517→        return []
   518→   518→
   519→   519→    def _parse_pull_response(self, data: dict) -> list[FetchedBlob]:
   520→   520→        """Parse the JSON response from a /pull request.
   521→   521→
   522→   522→        Args:
   523→   523→            data: Parsed JSON response from the relay.
   524→   524→
   525→   525→        Returns:
   526→   526→            List of FetchedBlob objects.
   527→   527→        """
   528→   528→        blobs: list[FetchedBlob] = []
   529→   529→        for item in data.get("blobs", []):
   530→   530→            blob_id = bytes.fromhex(item["blob_id"])
   531→   531→            encrypted_data = bytes.fromhex(item["encrypted_data"])
   532→   532→            hmac_tag = bytes.fromhex(item["hmac_tag"])
   533→   533→            expires_at = item.get("expires_at", 0.0)
   534→   534→
   535→   535→            blobs.append(FetchedBlob(
   536→   536→                blob_id=blob_id,
   537→   537→                encrypted_data=encrypted_data,
   538→   538→                hmac_tag=hmac_tag,
   539→   539→                expires_at=expires_at,
   540→   540→            ))
   541→   541→        return blobs

   544→   544→    # Local decryption
   545→   545→    # -------------------------------------------------------------------
   546→   546→
   547→   547→    def _decrypt_blob(self, blob: FetchedBlob) -> DecryptedNotification:
   548→   548→        """Decrypt a fetched blob using the relay-level key.
   549→   549→
   550→   550→        Note: The blob contains data that is already E2EE between
   551→   551→        the sender and recipient (via Double Ratchet or HPKE).
   552→   552→        The relay adds a layer of AES-256-GCM encryption using
   553→   553→        a key derived from the recipient's public key and the
   554→   554→        blob_id. This function strips that outer layer.
   555→   555→
   556→   556→        The inner payload is still encrypted and must be processed
   557→   557→        by the messaging layer (ChatManager / SealedSenderManager).
   558→   558→
   559→   559→        Args:
   560→   560→            blob: The fetched blob to decrypt.
   561→   561→
   562→   562→        Returns:
   563→   563→            DecryptedNotification with the inner encrypted payload.
   564→   564→
   565→   565→        Raises:
   566→   566→            DecryptionError: If the outer relay encryption cannot be removed.
   567→   567→        """
   568→   568→        # Derive relay-level decryption key from our public key + blob_id
   569→   569→        # Key = HKDF-SHA256(SHA-256(pk || blob_id), info="presidium-relay-key")
   570→   570→        combined = self._config.public_key + blob.blob_id
   571→   571→        prk = hashlib.sha256(combined).digest()
   572→   572→
   573→   573→        from cryptography.hazmat.primitives.kdf.hkdf import HKDF
   574→   574→        from cryptography.hazmat.primitives import hashes
   575→   575→
   576→   576→        hkdf = HKDF(
   577→   577→            algorithm=hashes.SHA256(),
   578→   578→            length=32,
   579→   579→            salt=blob.blob_id,
   580→   580→            info=b"presidium-relay-key",
   581→   581→        )
   582→   582→        relay_key = hkdf.derive(prk)
   583→   583→
   584→   584→        try:
   585→   585→            # Decrypt: first 12 bytes = nonce, rest = ciphertext + tag
   586→   586→            plaintext = AES256GCM.decrypt(
   587→   587→                key=relay_key,
   588→   588→                data=blob.encrypted_data,
   589→   589→                aad=blob.blob_id,
   590→   590→            )
   591→   591→        except Exception as exc:
   592→   592→            raise DecryptionError(
   593→   593→                f"Failed to decrypt relay-level encryption: {exc}"
   594→   594→            ) from exc
   595→   595→        finally:
   596→   596→            secure_zero(bytearray(relay_key))
   597→   597→
   598→   598→        # Parse the inner payload format:
   599→   599→        # [1B type_len][N bytes message_type]
   600→   600→        # [16B sender_user_id]
   601→   601→        # [rest: encrypted_payload]
   602→   602→        if len(plaintext) < 1 + 16:
   603→   603→            raise DecryptionError("Decrypted payload too short.")
   604→   604→
   605→   605→        offset = 0
   606→   606→        type_len = plaintext[offset]
   607→   607→        offset += 1
   608→   608→
   609→   609→        if len(plaintext) < offset + type_len + 16:
   610→   610→            raise DecryptionError("Decrypted payload truncated.")
   611→   611→
   612→   612→        message_type = plaintext[offset:offset + type_len].decode("utf-8")
   613→   613→        offset += type_len
   614→   614→
   615→   615→        sender_user_id = plaintext[offset:offset + 16].hex()
   616→   616→        offset += 16
   617→   617→
   618→   618→        encrypted_payload = plaintext[offset:]
   619→   619→
   620→   620→        return DecryptedNotification(
   621→   621→            sender_user_id=sender_user_id,
   622→   622→            message_type=message_type,
   623→   623→            encrypted_payload=encrypted_payload,
   624→   624→            received_at=time.time(),
   625→   625→        )
   626→   626→

   628→   628→    # Push to relay (sender side)
   629→   629→    # -------------------------------------------------------------------
   630→   630→
   631→   631→    async def push_to_relay(
   632→   632→        self,
   633→   633→        recipient_public_key: bytes,
   634→   634→        encrypted_payload: bytes,
   635→   635→        message_type: str = "chat",
   636→   636→        sender_user_id: str = "",
   637→   637→    ) -> str:
   638→   638→        """Push an encrypted blob to the relay for a recipient.
   639→   639→
   640→   640→        This is used by the SENDER to deposit an encrypted blob on the
   641→   641→        relay. The blob_id is computed from the recipient's public key
   642→   642→        and the current time bucket, so the recipient can find it.
   643→   643→
   644→   644→        Args:
   645→   645→            recipient_public_key: Recipient's X25519 public key (32 bytes).
   646→   646→            encrypted_payload: E2EE message content (already encrypted
   647→   647→                with Double Ratchet or HPKE).
   648→   648→            message_type: Type string for the notification.
   649→   649→            sender_user_id: Our user ID (included in relay-level payload).
   650→   650→
   651→   651→        Returns:
   652→   652→            Hex string of the stored blob_id.
   653→   653→
   654→   654→        Raises:
   655→   655→            FetchError: If the HTTP request to the relay fails.
   656→   656→        """
   657→   657→        if self._session is None:
   658→   658→            raise FetchError("Client not started — no HTTP session.")
   659→   659→
   660→   660→        # Compute the recipient's blob ID
   661→   661→        recipient_computer = BlobIDComputer(recipient_public_key)
   662→   662→        blob_id = recipient_computer.compute_current_ids(num_buckets=1)[0]
   663→   663→
   664→   664→        # Add relay-level encryption
   665→   665→        # Inner format: [1B type_len][type][16B sender_uid][payload]
   666→   666→        type_bytes = message_type.encode("utf-8")
   667→   667→        sender_uid_bytes = bytes.fromhex(sender_user_id) if sender_user_id else b"\x00" * 16
   668→   668→
   669→   669→        inner = (
   670→   670→            bytes([len(type_bytes)])
   671→   671→            + type_bytes
   672→   672→            + sender_uid_bytes
   673→   673→            + encrypted_payload
   674→   674→        )
   675→   675→
   676→   676→        # Derive relay-level encryption key
   677→   677→        combined = recipient_public_key + blob_id
   678→   678→        prk = hashlib.sha256(combined).digest()
   679→   679→
   680→   680→        from cryptography.hazmat.primitives.kdf.hkdf import HKDF
   681→   681→        from cryptography.hazmat.primitives import hashes
   682→   682→
   683→   683→        hkdf = HKDF(
   684→   684→            algorithm=hashes.SHA256(),
   685→   685→            length=32,
   686→   686→            salt=blob_id,
   687→   687→            info=b"presidium-relay-key",
   688→   688→        )
   689→   689→        relay_key = hkdf.derive(prk)
   690→   690→
   691→   691→        try:
   692→   692→            encrypted_data = AES256GCM.encrypt(
   693→   693→                key=relay_key,
   694→   694→                plaintext=inner,
   695→   695→                aad=blob_id,
   696→   696→            )
   697→   697→        finally:
   698→   698→            secure_zero(bytearray(relay_key))
   699→   699→
   700→   700→        # Compute HMAC
   701→   701→        import hmac as hmac_mod
   702→   702→        hmac_tag = hmac_mod.new(
   703→   703→            b"presidium-push-hmac",  # Client-side HMAC key (relay has its own)
   704→   704→            blob_id + encrypted_data,
   705→   705→            hashlib.sha256,
   706→   706→        ).digest()
   707→   707→
   708→   708→        # Build binary payload: [16B blob_id][4B data_len][N data][32B hmac]
   709→   709→        import struct as struct_mod
   710→   710→        payload = (
   711→   711→            blob_id
   712→   712→            + struct_mod.pack("!I", len(encrypted_data))
   713→   713→            + encrypted_data
   714→   714→            + hmac_tag
   715→   715→        )
   716→   716→
   717→   717→        # POST to relay
   718→   718→        url = f"{self._config.relay_url}/push"
   719→   719→        async with self._session.post(url, data=payload) as resp:
   720→   720→            if resp.status not in (200, 202):
   721→   721→                body = await resp.text()
   722→   722→                raise FetchError(f"Push failed with {resp.status}: {body[:200]}")
   723→   723→
   724→   724→        blob_id_hex = blob_id.hex()
   725→   725→        logger.debug("Pushed blob %s for recipient.", blob_id_hex)
   726→   726→        return blob_id_hex

   733→   733→        """Register our FCM/APNS device token with the relay."""
   734→   734→        if self._config.provider == PushProvider.FCM and self._config.fcm_token:
   735→   735→            await self.register_device_token(
   736→   736→                device_token=self._config.fcm_token,
   737→   737→                platform="fcm",
   738→   738→            )
   739→   739→        elif self._config.provider == PushProvider.APNS and self._config.apns_token:
   740→   740→            await self.register_device_token(
   741→   741→                device_token=self._config.apns_token,
   742→   742→                platform="apns",
   743→   743→            )
   744→   744→
   745→   745→    async def register_device_token(
   746→   746→        self, device_token: str, platform: str
   747→   747→    ) -> None:
   748→   748→        """Register a device token with the relay for push fallback.
   749→   749→
   750→   750→        The encrypted_user_id is derived from our public key so the
   751→   751→        relay can map push events to the right device without knowing
   752→   752→        our actual identity.
   753→   753→
   754→   754→        Args:
   755→   755→            device_token: FCM or APNS device token string.
   756→   756→            platform: "fcm" or "apns".
   757→   757→
   758→   758→        Raises:
   759→   759→            RegistrationError: If registration fails.
   760→   760→        """
   761→   761→        if self._session is None:
   762→   762→            raise RegistrationError("Client not started — no HTTP session.")
   763→   763→
   764→   764→        if platform not in ("fcm", "apns"):
   765→   765→            raise RegistrationError(f"Invalid platform: {platform!r}")
   766→   766→
   767→   767→        # Encrypt user ID for relay storage
   768→   768→        encrypted_uid = hashlib.sha256(
   769→   769→            self._config.public_key + b"presidium-device-uid"
   770→   770→        ).digest()
   771→   771→
   772→   772→        payload = {
   773→   773→            "encrypted_user_id": encrypted_uid.hex(),
   774→   774→            "device_token": device_token,
   775→   775→            "platform": platform,
   776→   776→        }
   777→   777→
   778→   778→        url = f"{self._config.relay_url}/register"
   779→   779→        try:
   780→   780→            async with self._session.post(url, json=payload) as resp:
   781→   781→                if resp.status not in (200, 201):
   782→   782→                    body = await resp.text()
   783→   783→                    raise RegistrationError(
   784→   784→                        f"Registration failed with {resp.status}: {body[:200]}"
   785→   785→                    )
   786→   786→        except aiohttp.ClientError as exc:
   787→   787→            raise RegistrationError(f"Registration request failed: {exc}") from exc
   788→   788→
   789→   789→        logger.info("Registered %s device token with relay.", platform.upper())
   790→   790→

   792→   792→        """Unregister a device token from the relay.
   793→   793→
   794→   794→        Args:
   795→   795→            device_token: The token to unregister.
