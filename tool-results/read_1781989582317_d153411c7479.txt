   172→# ---------------------------------------------------------------------------
   173→# Key generation worker (background thread)
   174→# ---------------------------------------------------------------------------
   175→
   176→class KeyGenWorker(QThread):
   177→    """Background thread for cryptographic key pair generation.
   178→
   179→    Generates an Ed25519 identity and emits the user_id on success
   180→    or an error message on failure.
   181→    """
   182→
   183→    finished = Signal(str)   # user_id
   184→    progress = Signal(int)   # percentage 0–100
   185→    error = Signal(str)
   186→
   187→    def __init__(self, identity_manager, password: str, nickname: str) -> None:
   188→        super().__init__()
   189→        self._identity_manager = identity_manager
   190→        self._password = password
   191→        self._nickname = nickname
   192→
   193→    def run(self) -> None:
   194→        """Generate identity in background thread with progress updates."""
   195→        loop = asyncio.new_event_loop()
   196→        try:
   197→            self.progress.emit(10)
   198→            # Simulate staged key generation progress
   199→            import time
   200→            time.sleep(0.1)  # Brief pause for UX feedback
   201→            self.progress.emit(30)
   202→
   203→            user_id = loop.run_until_complete(
   204→                self._identity_manager.create_identity(
   205→                    self._password,
   206→                    self._nickname or None,
   207→                )
   208→            )
   209→            self.progress.emit(80)
   210→            time.sleep(0.05)
   211→            self.progress.emit(100)
   212→            self.finished.emit(user_id)
   213→        except Exception as exc:
   214→            self.error.emit(str(exc))
   215→        finally:
   216→            loop.close()
   217→
   218→
   219→# ---------------------------------------------------------------------------
   220→# Recovery phrase generator
   221→# ---------------------------------------------------------------------------
   222→
   223→def generate_recovery_phrase(word_count: int = RECOVERY_WORD_COUNT) -> list[str]:
   224→    """Generate a BIP39-style recovery phrase.
   225→
   226→    Uses cryptographically secure random selection from the word list.
   227→    The entropy source is ``secrets.randbelow()``.
   228→
   229→    Args:
   230→        word_count: Number of words in the phrase (default 12).
   231→
   232→    Returns:
   233→        List of words forming the recovery phrase.
   234→    """
   235→    return [_BIP39_WORDS[secrets.randbelow(len(_BIP39_WORDS))] for _ in range(word_count)]

   305→    def _setup_step_indicator(self) -> None:
   306→        """Create the horizontal step-progress dots."""
   307→        layout = QHBoxLayout(self._step_indicator)
   308→        layout.setContentsMargins(24, 8, 24, 8)
   309→        layout.setSpacing(8)
   310→
   311→        self._step_dots: list[QLabel] = []
   312→        for i in range(5):
   313→            dot = QLabel()
   314→            dot.setFixedSize(8, 8)
   315→            dot.setStyleSheet("background-color: #3A3A50; border-radius: 4px;")
   316→            dot.setAlignment(Qt.AlignmentFlag.AlignCenter)
   317→            self._step_dots.append(dot)
   318→            layout.addWidget(dot)
   319→
   320→        layout.addStretch()
   321→        self._update_step_indicator()
   322→
   323→    def _update_step_indicator(self) -> None:
   324→        """Highlight the current step dot."""
   325→        for i, dot in enumerate(self._step_dots):
   326→            if i == self._current_step:
   327→                dot.setStyleSheet(
   328→                    "background-color: #6366F1; border-radius: 4px; "
   329→                    "min-width: 24px; max-width: 24px;"
   330→                )
   331→                dot.setFixedSize(24, 8)
   332→            elif i < self._current_step:
   333→                dot.setStyleSheet("background-color: #8B8BA8; border-radius: 4px;")
   334→                dot.setFixedSize(8, 8)
   335→            else:
   336→                dot.setStyleSheet("background-color: #3A3A50; border-radius: 4px;")
   337→                dot.setFixedSize(8, 8)

   825→    def _validate_password(self) -> str:
   826→        """Validate password fields and return the password.
   827→
   828→        Returns:
   829→            The validated password string.
   830→
   831→        Raises:
   832→            WeakPasswordError: If the password is too short.
   833→            PasswordMismatchError: If the passwords don't match.
   834→        """
   835→        password = self._password_input.text()
   836→        confirm = self._confirm_input.text()
   837→
   838→        if len(password) < 8:
   839→            raise WeakPasswordError(_("Password must be at least 8 characters."))
   840→
   841→        if password != confirm:
   842→            raise PasswordMismatchError(_("Passwords do not match."))
   843→
   844→        # Basic strength check: not all same character
   845→        if len(set(password)) < 3:
   846→            raise WeakPasswordError(_("Password is too weak. Use a mix of characters."))
   847→
   848→        return password
   849→
   850→    @Slot()
   851→    def _on_generate_keys(self) -> None:
   852→        """Validate and start key generation."""
   853→        try:
   854→            password = self._validate_password()
   855→        except (WeakPasswordError, PasswordMismatchError) as exc:
   856→            self._error_label.setText(str(exc))
   857→            return
   858→
   859→        self._error_label.setText("")
   860→        self._progress.setVisible(True)
   861→        self._progress_label.setText(_("Generating Ed25519 key pair..."))
   862→        self._generate_btn.setEnabled(False)
   863→
   864→        nickname = self._nickname_input.text() if hasattr(self, "_nickname_input") else ""
   865→
   866→        if self._identity_manager is not None:
   867→            self._worker = KeyGenWorker(self._identity_manager, password, nickname)
   868→            self._worker.progress.connect(self._on_key_gen_progress)
   869→            self._worker.finished.connect(self._on_key_gen_success)
   870→            self._worker.error.connect(self._on_key_gen_error)
   871→            self._worker.start()
   872→        else:
   873→            # Standalone / test mode: simulate key generation
   874→            self._simulate_key_generation(password)
   875→
   876→    @Slot(int)
   877→    def _on_key_gen_progress(self, percent: int) -> None:
   878→        """Update the progress bar during key generation."""
   879→        self._progress.setValue(percent)
   880→        if percent < 50:
   881→            self._progress_label.setText(_("Generating Ed25519 key pair..."))
   882→        elif percent < 80:
   883→            self._progress_label.setText(_("Encrypting with Argon2id..."))
   884→        else:
   885→            self._progress_label.setText(_("Finalizing identity..."))
   886→
   887→    @Slot(str)
   888→    def _on_key_gen_success(self, user_id: str) -> None:
   889→        """Handle successful key generation."""
   890→        self._user_id = user_id
   891→        self._progress.setValue(100)
   892→        self._progress_label.setText(_("Identity created successfully!"))
   893→
   894→        # Generate recovery phrase
   895→        self._recovery_phrase = generate_recovery_phrase()
   896→        self._populate_recovery_grid()
   897→
   898→        # Navigate to recovery step
   899→        self._go_to_step(WizardStep.RECOVERY)
   900→
   901→    @Slot(str)
   902→    def _on_key_gen_error(self, error_msg: str) -> None:
   903→        """Handle key generation failure."""
   904→        self._error_label.setText(_("Error: {error}").format(error=error_msg))
   905→        self._progress.setVisible(False)
   906→        self._progress_label.setText("")
   907→        self._generate_btn.setEnabled(True)
   908→
   909→    def _simulate_key_generation(self, password: str) -> None:
   910→        """Simulate key generation for standalone / test mode.
   911→
   912→        Creates a deterministic user_id from the password hash.
   913→
   914→        Args:
   915→            password: The user's password.
   916→        """
   917→        from PySide6.QtCore import QTimer
   918→
   919→        self._progress.setValue(30)
   920→        self._progress_label.setText(_("Generating Ed25519 key pair..."))
   921→
   922→        def _step2() -> None:
   923→            self._progress.setValue(60)
   924→            self._progress_label.setText(_("Encrypting with Argon2id..."))
   925→
   926→        def _step3() -> None:
   927→            self._progress.setValue(90)
   928→            self._progress_label.setText(_("Finalizing identity..."))
   929→
   930→        def _done() -> None:
   931→            # Create a fake user_id from the password
   932→            fake_uid = hashlib.sha256(password.encode()).hexdigest()[:32]
   933→            self._user_id = fake_uid
   934→            self._progress.setValue(100)
   935→            self._progress_label.setText(_("Identity created successfully!"))
   936→            self._recovery_phrase = generate_recovery_phrase()
   937→            self._populate_recovery_grid()
   938→            self._go_to_step(WizardStep.RECOVERY)
   939→
   940→        QTimer.singleShot(300, _step2)
   941→        QTimer.singleShot(600, _step3)
   942→        QTimer.singleShot(900, _done)
   943→
