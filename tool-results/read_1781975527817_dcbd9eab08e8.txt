     1→"""Import module for migrating from other messengers into Presidium.
     2→
     3→Provides a complete migration pipeline for importing conversation history
     4→and contacts from Signal, WhatsApp, Telegram, and Matrix. Each source has
     5→a dedicated converter that understands the export format, validates and
     6→sanitizes the data, then inserts it into the Presidium database.
     7→
     8→Architecture:
     9→    ImportManager (orchestrator)
    10→        ├── MessageConverter (abstract base)
    11→        │       ├── SignalConverter  — reads Signal JSON exports
    12→        │       ├── WhatsAppConverter — reads WhatsApp .txt exports
    13→        │       ├── TelegramConverter — reads Telegram JSON exports
    14→        │       └── (extensible for future sources)
    15→        ├── ValidationPipeline — sanitizes and validates every message
    16→        ├── ProgressReporter — optional callback for UI progress bars
    17→        └── ImportResult — summary dataclass for the import outcome
    18→
    19→Import flow:
    20→    1. User selects export directory/file
    21→    2. ImportManager detects source type and instantiates the correct Converter
    22→    3. Converter reads and parses the export format into intermediate records
    23→    4. Each record is validated (XSS, injection, length limits, encoding)
    24→    5. Validated records are inserted into the Presidium DB in batches
    25→    6. Progress callbacks fire after each batch
    26→    7. ImportResult summarizes total / imported / skipped / errors
    27→
    28→Security considerations:
    29→    - All imported text is sanitized (no HTML/JS injection)
    30→    - Attachments are quarantined and scanned before linking
    31→    - Sender identities are mapped to Presidium contact format
    32→    - No raw external data enters the DB without validation
    33→
    34→Supported export formats:
    35→    - Signal: JSON export (signal-export format)
    36→    - WhatsApp: _chat.txt (WhatsApp chat export format)
    37→    - Telegram: result.json (Telegram Desktop export)
    38→    - Matrix: JSON export (matrix-archive format, planned)
    39→"""

   228→# ---------------------------------------------------------------------------
   229→# Validation pipeline
   230→# ---------------------------------------------------------------------------
   231→
   232→class ValidationPipeline:
   233→    """Sanitizes and validates imported messages before database insertion.
   234→
   235→    Applies a sequence of validation rules to each ImportedMessage:
   236→        1. Text sanitization (strip HTML tags, control characters)
   237→        2. Length limits enforcement
   238→        3. Sender name normalization
   239→        4. Timestamp range verification
   240→        5. Attachment validation
   241→
   242→    Messages that fail validation are either corrected (best-effort) or
   243→    flagged as skipped.
   244→    """
   245→
   246→    @staticmethod
   247→    def sanitize_text(text: str) -> str:
   248→        """Remove HTML tags and control characters from message text.
   249→
   250→        Args:
   251→            text: Raw message text from the export.
   252→
   253→        Returns:
   254→            Sanitized text safe for storage and display.
   255→        """
   256→        cleaned = SANITIZE_HTML_PATTERN.sub("", text)
   257→        cleaned = SANITIZE_CONTROL_CHARS.sub("", cleaned)
   258→        return cleaned.strip()
   259→
   260→    @staticmethod
   261→    def truncate_text(text: str, max_length: int = MAX_MESSAGE_LENGTH) -> str:
   262→        """Truncate text to the maximum allowed length.
   263→
   264→        Args:
   265→            text: Message text to potentially truncate.
   266→            max_length: Maximum number of characters allowed.
   267→
   268→        Returns:
   269→            Text truncated to max_length if necessary.
   270→        """
   271→        if len(text) <= max_length:
   272→            return text
   273→        return text[:max_length]
   274→
   275→    @staticmethod
   276→    def sanitize_sender_name(name: str) -> str:
   277→        """Sanitize and truncate a sender display name.
   278→
   279→        Args:
   280→            name: Raw sender name from the export.
   281→
   282→        Returns:
   283→            Cleaned sender name within length limits.
   284→        """
   285→        cleaned = SANITIZE_CONTROL_CHARS.sub("", name).strip()
   286→        if len(cleaned) > MAX_SENDER_NAME_LENGTH:
   287→            cleaned = cleaned[:MAX_SENDER_NAME_LENGTH]
   288→        return cleaned
   289→
   290→    @staticmethod
   291→    def validate_timestamp(timestamp: float) -> float:
   292→        """Validate a message timestamp is within reasonable bounds.
   293→
   294→        Rejects timestamps before 2000-01-01 or in the future.
   295→
   296→        Args:
   297→            timestamp: Unix timestamp in seconds.
   298→
   299→        Returns:
   300→            The validated timestamp.
   301→
   302→        Raises:
   303→            ImportValidationError: If the timestamp is out of range.
   304→        """
   305→        min_timestamp = 946684800.0  # 2000-01-01 00:00:00 UTC
   306→        max_timestamp = time.time() + 86400.0  # Allow up to 1 day in future
   307→        if timestamp < min_timestamp:
   308→            raise ImportValidationError(
   309→                _("Timestamp {ts} is before year 2000, likely invalid.").format(
   310→                    ts=timestamp
   311→                )
   312→            )
   313→        if timestamp > max_timestamp:
   314→            raise ImportValidationError(
   315→                _("Timestamp {ts} is in the future, likely invalid.").format(
   316→                    ts=timestamp
   317→                )
   318→            )
   319→        return timestamp
   320→
   321→    def validate_message(self, message: ImportedMessage) -> ImportedMessage:
   322→        """Run the full validation pipeline on a single message.
   323→
   324→        Args:
   325→            message: The imported message to validate.
   326→
   327→        Returns:
   328→            The validated and sanitized message.
   329→
   330→        Raises:
   331→            ImportValidationError: If the message cannot be salvaged.
   332→        """
   333→        # Sanitize text content
   334→        message.text = self.sanitize_text(message.text)
   335→        message.text = self.truncate_text(message.text)
   336→
   337→        # Sanitize sender name
   338→        message.sender_name = self.sanitize_sender_name(message.sender_name)
   339→
   340→        # Validate timestamp
   341→        if message.timestamp > 0:
   342→            message.timestamp = self.validate_timestamp(message.timestamp)
   343→        else:
   344→            # Default to epoch if missing; allow through but log
   345→            logger.debug(
   346→                "Message with external_id=%s has no timestamp, defaulting to 0.",
   347→                message.external_id,
   348→            )
   349→
   350→        # Validate attachments
   351→        validated_attachments: list[ImportedAttachment] = []
   352→        for attachment in message.attachments:
   353→            if attachment.file_name:
   354→                attachment.file_name = self.sanitize_text(attachment.file_name)
   355→            if attachment.original_path:
   356→                # Strip path traversal attempts
   357→                attachment.original_path = attachment.original_path.replace(
   358→                    "..", ""
   359→                ).lstrip("/\\")
   360→            validated_attachments.append(attachment)
   361→        message.attachments = validated_attachments
   362→
   363→        return message

   370→class MessageConverter(abc.ABC):
   371→    """Abstract base class for messenger-specific export converters.
   372→
   373→    Each subclass understands a particular export format and produces
   374→    a sequence of ImportedMessage objects ready for validation and
   375→    database insertion.
   376→
   377→    Subclasses must implement:
   378→        - parse(): Read the export file and yield ImportedMessage objects
   379→        - detect_format(): Check if a given path matches this converter's format
   380→    """
   381→
   382→    @abc.abstractmethod
   383→    async def parse(
   384→        self,
   385→        export_path: Path,
   386→        progress_callback: Optional[ProgressCallback] = None,
   387→    ) -> list[ImportedMessage]:
   388→        """Parse an export file into a list of imported messages.
   389→
   390→        Args:
   391→            export_path: Path to the export file or directory.
   392→            progress_callback: Optional callback for progress reporting.
   393→
   394→        Returns:
   395→            List of parsed ImportedMessage objects.
   396→
   397→        Raises:
   398→            ImportFormatError: If the export file cannot be parsed.
   399→        """
   400→
   401→    @abc.abstractmethod
   402→    def detect_format(self, export_path: Path) -> bool:
   403→        """Check if the given path matches this converter's export format.
   404→
   405→        Args:
   406→            export_path: Path to the export file or directory.
   407→
   408→        Returns:
   409→            True if this converter can handle the export.
   410→        """
   411→

   413→# ---------------------------------------------------------------------------
   414→# Signal converter

   416→
   417→class SignalConverter(MessageConverter):
   418→    """Converts Signal JSON exports to Presidium ImportedMessage objects.
   419→
   420→    Signal exports are produced by signal-export and similar tools.
   421→    The JSON format is a list of conversations, each containing messages:
   422→
   423→        [
   424→            {
   425→                "name": "Contact Name",
   426→                "type": "private" | "group",
   427→                "messages": [
   428→                    {
   429→                        "timestamp": 1700000000000,
   430→                        "sender": "Me" | "Contact Name",
   431→                        "body": "Hello!",
   432→                        "attachments": [{"path": "attachments/abc.jpg", "contentType": "image/jpeg"}],
   433→                        "quote": { "id": 123, "author": "..." },
   434→                        "type": "incoming" | "outgoing"
   435→                    }
   436→                ]
   437→            }
   438→        ]
   439→
   440→    Signal timestamps are in milliseconds since epoch.
   441→    """
   442→
   443→    def detect_format(self, export_path: Path) -> bool:
   444→        """Detect Signal export by checking for JSON with Signal-specific keys."""
   445→        if export_path.is_dir():
   446→            json_files = list(export_path.glob("*.json"))
   447→            if not json_files:
   448→                return False
   449→            export_path = json_files[0]
   450→
   451→        if not export_path.is_file() or export_path.suffix != ".json":
   452→            return False
   453→
   454→        try:
   455→            with open(export_path, "r", encoding="utf-8") as f:
   456→                data = json.load(f)
   457→            if isinstance(data, list) and len(data) > 0:
   458→                first = data[0]
   459→                return "messages" in first and isinstance(first.get("messages"), list)
   460→        except (json.JSONDecodeError, OSError, KeyError):
   461→            return False
   462→
   463→        return False
   464→
   465→    async def parse(
   466→        self,
   467→        export_path: Path,
   468→        progress_callback: Optional[ProgressCallback] = None,
   469→    ) -> list[ImportedMessage]:
   470→        """Parse a Signal JSON export.
   471→
   472→        Args:
   473→            export_path: Path to the Signal export JSON file.
   474→            progress_callback: Optional progress callback.
   475→
   476→        Returns:
   477→            List of ImportedMessage objects.
   478→
   479→        Raises:
   480→            ImportFormatError: If the file cannot be parsed.
   481→        """
   482→        if export_path.is_dir():
   483→            json_files = list(export_path.glob("*.json"))
   484→            if not json_files:
   485→                raise ImportFormatError(
   486→                    _("No JSON files found in Signal export directory: {path}").format(
   487→                        path=export_path
   488→                    )
   489→                )
   490→            export_path = json_files[0]
   491→
   492→        try:
   493→            with open(export_path, "r", encoding="utf-8") as f:
   494→                data = json.load(f)
   495→        except json.JSONDecodeError as exc:
   496→            raise ImportFormatError(
   497→                _("Invalid JSON in Signal export: {error}").format(error=exc)
   498→            ) from exc
   499→        except OSError as exc:
   500→            raise ImportFormatError(
   501→                _("Cannot read Signal export file: {error}").format(error=exc)
   502→            ) from exc
   503→
   504→        if not isinstance(data, list):
   505→            raise ImportFormatError(
   506→                _("Signal export must be a JSON array of conversations.")
   507→            )
   508→
   509→        messages: list[ImportedMessage] = []
   510→        total_conversations = len(data)
   511→
   512→        for conv_idx, conversation in enumerate(data):
   513→            conv_name = conversation.get("name", _("Unknown Chat"))
   514→            conv_messages = conversation.get("messages", [])
   515→
   516→            for raw_msg in conv_messages:
   517→                try:
   518→                    msg = self._convert_signal_message(
   519→                        raw_msg, conv_name, export_path.parent
   520→                    )
   521→                    messages.append(msg)
   522→                except (ImportValidationError, KeyError, TypeError) as exc:
   523→                    logger.debug(
   524→                        "Skipping invalid Signal message in %s: %s",
   525→                        conv_name, exc,
   526→                    )
   527→
   528→            if progress_callback is not None:
   529→                progress_callback(conv_idx + 1, total_conversations, ImportSource.SIGNAL)
   530→
   531→            # Yield control periodically for large exports
   532→            if conv_idx % 10 == 0:
   533→                await asyncio.sleep(0)
   534→
   535→        logger.info(
   536→            "Parsed %d messages from %d Signal conversations.",
   537→            len(messages), total_conversations,
   538→        )
   539→        return messages
   540→
   541→    def _convert_signal_message(
   542→        self,
   543→        raw: dict[str, Any],
   544→        chat_name: str,
   545→        base_dir: Path,
   546→    ) -> ImportedMessage:
   547→        """Convert a single Signal message dict to ImportedMessage.
   548→
   549→        Args:
   550→            raw: Raw message dict from Signal JSON.
   551→            chat_name: Name of the conversation.
   552→            base_dir: Base directory for resolving attachment paths.
   553→
   554→        Returns:
   555→            ImportedMessage object.
   556→
   557→        Raises:
   558→            ImportValidationError: If required fields are missing.
   559→        """
   560→        # Signal uses millisecond timestamps
   561→        ts_ms = raw.get("timestamp", 0)
   562→        timestamp = ts_ms / 1000.0 if isinstance(ts_ms, (int, float)) else 0.0
   563→
   564→        sender = raw.get("sender", "")
   565→        msg_type = raw.get("type", "incoming")
   566→        is_outgoing = msg_type == "outgoing"
   567→
   568→        body = raw.get("body", "")
   569→        if body is None:
   570→            body = ""
   571→
   572→        # Extract attachments
   573→        attachments: list[ImportedAttachment] = []
   574→        for att in raw.get("attachments", []):
   575→            att_path = att.get("path", "")
   576→            content_type = att.get("contentType", "application/octet-stream")
   577→            media_type = self._mime_to_media_type(content_type)
   578→            file_name = att_path.split("/")[-1] if att_path else ""
   579→            attachments.append(ImportedAttachment(
   580→                media_type=media_type,
   581→                original_path=str(base_dir / att_path) if att_path else "",
   582→                file_name=file_name,
   583→                mime_type=content_type,
   584→                size_bytes=0,
   585→            ))
   586→
   587→        # Extract reply-to reference
   588→        quote = raw.get("quote")
   589→        reply_to_id = ""
   590→        if isinstance(quote, dict):
   591→            reply_to_id = str(quote.get("id", ""))
   592→
   593→        return ImportedMessage(
   594→            source=ImportSource.SIGNAL,
   595→            external_id=str(raw.get("timestamp", uuid.uuid4().hex)),
   596→            sender_name=sender,
   597→            sender_id="",
   598→            timestamp=timestamp,
   599→            text=body,
   600→            attachments=attachments,
   601→            reply_to_id=reply_to_id,
   602→            is_outgoing=is_outgoing,
   603→            chat_name=chat_name,
   604→            raw_data=raw,
   605→        )
   606→
   607→    @staticmethod
   608→    def _mime_to_media_type(mime_type: str) -> MessageMediaType:
   609→        """Map a MIME type to a MessageMediaType enum value.
   610→
   611→        Args:
   612→            mime_type: MIME type string (e.g., "image/jpeg").
   613→
   614→        Returns:
   615→            Corresponding MessageMediaType value.
   616→        """
   617→        if mime_type.startswith("image/"):
   618→            return MessageMediaType.IMAGE
   619→        if mime_type.startswith("video/"):
   620→            return MessageMediaType.VIDEO
   621→        if mime_type.startswith("audio/"):
   622→            return MessageMediaType.AUDIO
   623→        if "sticker" in mime_type:
   624→            return MessageMediaType.STICKER
   625→        if "voice" in mime_type:
   626→            return MessageMediaType.VOICE
   627→        if "animation" in mime_type or mime_type == "image/gif":
   628→            return MessageMediaType.ANIMATION
   629→        return MessageMediaType.FILE

   635→
   636→class WhatsAppConverter(MessageConverter):
   637→    """Converts WhatsApp .txt chat exports to Presidium ImportedMessage objects.
   638→
   639→    WhatsApp exports are plain text files (_chat.txt) with one message per line:
   640→
   641→        12/25/2023, 10:30 AM - John: Hello!
   642→        12/25/2023, 10:31 AM - Jane: Hi there!
   643→        12/25/2023, 10:32 AM - John: <attached: photo.jpg>
   644→        12/25/2023, 10:33 AM - Jane left
   645→        12/25/2023, 10:34 AM - You created group "Friends"
   646→
   647→    Line format (varies by locale):
   648→        MM/DD/YYYY, HH:MM AM/PM - Sender: Message text
   649→        DD/MM/YYYY, HH:MM - Sender: Message text
   650→
   651→    System messages (added to group, left, etc.) are captured but marked
   652→    as system events rather than user messages.
   653→    """
   654→
   655→    # Pattern for US-style date: 12/25/2023, 10:30 AM - Sender: text
   656→    PATTERN_US = re.compile(
   657→        r"(\d{1,2}/\d{1,2}/\d{2,4}),\s*(\d{1,2}:\d{2}\s*[AP]M)\s*-\s*"
   658→        r"([^:]+?):\s*(.*)",
   659→        re.DOTALL,
   660→    )
   661→
   662→    # Pattern for EU-style date: 25/12/2023, 10:30 - Sender: text
   663→    PATTERN_EU = re.compile(
   664→        r"(\d{1,2}/\d{1,2}/\d{2,4}),\s*(\d{1,2}:\d{2})\s*-\s*"
   665→        r"([^:]+?):\s*(.*)",
   666→        re.DOTALL,
   667→    )
   668→
   669→    # Pattern for system messages (no sender colon)
   670→    PATTERN_SYSTEM = re.compile(
   671→        r"(\d{1,2}/\d{1,2}/\d{2,4}),\s*(\d{1,2}:\d{2}(?:\s*[AP]M)?)\s*-\s*(.*)",
   672→    )
   673→
   674→    # Attachment indicators
   675→    ATTACHMENT_PATTERNS: list[tuple[re.Pattern[str], MessageMediaType]] = [
   676→        (re.compile(r"<attached:\s*.*\.jpg>", re.IGNORECASE), MessageMediaType.IMAGE),
   677→        (re.compile(r"<attached:\s*.*\.png>", re.IGNORECASE), MessageMediaType.IMAGE),
   678→        (re.compile(r"<attached:\s*.*\.mp4>", re.IGNORECASE), MessageMediaType.VIDEO),
   679→        (re.compile(r"<attached:\s*.*\.opus>", re.IGNORECASE), MessageMediaType.VOICE),
   680→        (re.compile(r"<attached:\s*.*\.pdf>", re.IGNORECASE), MessageMediaType.FILE),
   681→        (re.compile(r"IMG-\d{8}-WA\d+", re.IGNORECASE), MessageMediaType.IMAGE),
   682→        (re.compile(r"PTT-\d{8}-WA\d+", re.IGNORECASE), MessageMediaType.VOICE),
   683→        (re.compile(r"VID-\d{8}-WA\d+", re.IGNORECASE), MessageMediaType.VIDEO),
   684→        (re.compile(r"DOC-\d{8}-WA\d+", re.IGNORECASE), MessageMediaType.FILE),
   685→    ]
   686→
   687→    def detect_format(self, export_path: Path) -> bool:
   688→        """Detect WhatsApp export by checking for _chat.txt or similar files."""
   689→        if export_path.is_file():
   690→            name = export_path.name.lower()
   691→            if name.endswith(".txt") and "chat" in name:
   692→                return True
   693→
   694→        if export_path.is_dir():
   695→            chat_files = list(export_path.glob("*chat*.txt"))
   696→            return len(chat_files) > 0
   697→
   698→        return False

   700→    async def parse(
   701→        self,
   702→        export_path: Path,
   703→        progress_callback: Optional[ProgressCallback] = None,
   704→    ) -> list[ImportedMessage]:
   705→        """Parse a WhatsApp .txt chat export.
   706→
   707→        Args:
   708→            export_path: Path to the WhatsApp export file or directory.
   709→            progress_callback: Optional progress callback.
   710→
   711→        Returns:
   712→            List of ImportedMessage objects.
   713→
   714→        Raises:
   715→            ImportFormatError: If the file cannot be parsed.
   716→        """
   717→        chat_file = self._resolve_chat_file(export_path)
   718→
   719→        try:
   720→            with open(chat_file, "r", encoding="utf-8-sig") as f:
   721→                lines = f.readlines()
   722→        except OSError as exc:
   723→            raise ImportFormatError(
   724→                _("Cannot read WhatsApp export file: {error}").format(error=exc)
   725→            ) from exc
   726→
   727→        total_lines = len(lines)
   728→        messages: list[ImportedMessage] = []
   729→        current_message_lines: list[str] = []
   730→        line_idx = 0
   731→
   732→        for line_idx, line in enumerate(lines):
   733→            stripped = line.strip()
   734→            if not stripped:
   735→                continue
   736→
   737→            # Check if this line starts a new message (has date prefix)
   738→            if self._is_new_message_line(stripped):
   739→                # Flush the previous multi-line message
   740→                if current_message_lines:
   741→                    msg = self._parse_message_block(current_message_lines)
   742→                    if msg is not None:
   743→                        messages.append(msg)
   744→                current_message_lines = [stripped]
   745→            else:
   746→                # Continuation of the current message
   747→                current_message_lines.append(stripped)
   748→
   749→            if progress_callback is not None and line_idx % BATCH_SIZE == 0:
   750→                progress_callback(line_idx, total_lines, ImportSource.WHATSAPP)
   751→
   752→            # Yield control periodically
   753→            if line_idx % 500 == 0:
   754→                await asyncio.sleep(0)
   755→
   756→        # Flush final message
   757→        if current_message_lines:
   758→            msg = self._parse_message_block(current_message_lines)
   759→            if msg is not None:
   760→                messages.append(msg)
   761→
   762→        if progress_callback is not None:
   763→            progress_callback(total_lines, total_lines, ImportSource.WHATSAPP)
   764→
   765→        logger.info(
   766→            "Parsed %d messages from WhatsApp export (%d lines).",
   767→            len(messages), total_lines,
   768→        )
   769→        return messages
   770→
   771→    def _resolve_chat_file(self, export_path: Path) -> Path:
   772→        """Resolve the chat file path from a file or directory.
   773→
   774→        Args:
   775→            export_path: Path to file or directory.
   776→
   777→        Returns:
   778→            Resolved Path to the chat file.
   779→
   780→        Raises:
   781→            ImportFormatError: If no chat file can be found.
   782→        """
   783→        if export_path.is_file():
   784→            return export_path
   785→
   786→        chat_files = list(export_path.glob("*chat*.txt"))
   787→        if chat_files:
   788→            return chat_files[0]
   789→
   790→        # Try common WhatsApp export names
   791→        for name in ("_chat.txt", "WhatsApp Chat.txt", "chat.txt"):
   792→            candidate = export_path / name
   793→            if candidate.is_file():
   794→                return candidate
   795→
   796→        raise ImportFormatError(
   797→            _("No WhatsApp chat file found in: {path}").format(path=export_path)
   798→        )

   800→    def _is_new_message_line(self, line: str) -> bool:
   801→        """Check if a line starts a new message (has date/time prefix).
   802→
   803→        Args:
   804→            line: A single line from the export file.
   805→
   806→        Returns:
   807→            True if this line begins a new message entry.
   808→        """
   809→        return bool(self.PATTERN_SYSTEM.match(line))
   810→
   811→    def _parse_message_block(self, lines: list[str]) -> Optional[ImportedMessage]:
   812→        """Parse a multi-line message block into an ImportedMessage.
   813→
   814→        Args:
   815→            lines: Lines belonging to a single message.
   816→
   817→        Returns:
   818→            ImportedMessage or None if the block is a system message.
   819→        """
   820→        full_text = " ".join(lines)
   821→
   822→        # Try user message patterns
   823→        match = self.PATTERN_US.match(full_text)
   824→        if match is None:
   825→            match = self.PATTERN_EU.match(full_text)
   826→
   827→        if match is not None:
   828→            date_str, time_str, sender, body = match.groups()
   829→            timestamp = self._parse_whatsapp_datetime(date_str, time_str)
   830→            is_outgoing = sender.strip().lower() in ("you", _("You").lower())
   831→
   832→            # Detect attachments in body text
   833→            attachments = self._detect_attachments(body)
   834→
   835→            return ImportedMessage(
   836→                source=ImportSource.WHATSAPP,
   837→                external_id=uuid.uuid4().hex[:16],
   838→                sender_name=sender.strip(),
   839→                sender_id="",
   840→                timestamp=timestamp,
   841→                text=body.strip(),
   842→                attachments=attachments,
   843→                reply_to_id="",
   844→                is_outgoing=is_outgoing,
   845→                chat_name="",
   846→                raw_data={"date": date_str, "time": time_str, "full_line": full_text},
   847→            )
   848→
   849→        # System message (no sender colon) — skip but don't error
   850→        return None
   851→
   852→    @staticmethod
   853→    def _parse_whatsapp_datetime(date_str: str, time_str: str) -> float:
   854→        """Parse WhatsApp date/time strings into a Unix timestamp.
   855→
   856→        Args:
   857→            date_str: Date portion (e.g., "12/25/2023").
   858→            time_str: Time portion (e.g., "10:30 AM").
   859→
   860→        Returns:
   861→            Unix timestamp in seconds, or 0.0 if parsing fails.
   862→        """
   863→        import calendar
   864→        from datetime import datetime
   865→
   866→        try:
   867→            # Normalize time string
   868→            time_normalized = time_str.strip().upper()
   869→            if "AM" in time_normalized or "PM" in time_normalized:
   870→                fmt_time = "%I:%M %p"
   871→            else:
   872→                fmt_time = "%H:%M"
   873→
   874→            # Try MM/DD/YYYY first (US format)
   875→            try:
   876→                dt = datetime.strptime(
   877→                    f"{date_str} {time_normalized}", f"%m/%d/%Y {fmt_time}"
   878→                )
   879→                return dt.timestamp()
   880→            except ValueError:
   881→                logger.debug("US date format did not match: %s", date_str)
   882→
   883→            # Try DD/MM/YYYY (EU format)
   884→            try:
   885→                dt = datetime.strptime(
   886→                    f"{date_str} {time_normalized}", f"%d/%m/%Y {fmt_time}"
   887→                )
   888→                return dt.timestamp()
   889→            except ValueError:
   890→                logger.debug("EU date format did not match: %s", date_str)
   891→
   892→            # Try 2-digit year
   893→            for date_fmt in ("%m/%d/%y", "%d/%m/%y"):
   894→                try:
   895→                    dt = datetime.strptime(
   896→                        f"{date_str} {time_normalized}", f"{date_fmt} {fmt_time}"
   897→                    )
   898→                    return dt.timestamp()
   899→                except ValueError:
   900→                    continue
   901→
   902→        except Exception as exc:
   903→            logger.debug("Failed to parse WhatsApp datetime '%s %s': %s", date_str, time_str, exc)
   904→
   905→        return 0.0
   906→
   907→    def _detect_attachments(self, body: str) -> list[ImportedAttachment]:
   908→        """Detect attachment indicators in WhatsApp message body.
   909→
   910→        Args:
   911→            body: Message body text.
   912→
   913→        Returns:
   914→            List of ImportedAttachment objects for detected media.
   915→        """
   916→        attachments: list[ImportedAttachment] = []
   917→        for pattern, media_type in self.ATTACHMENT_PATTERNS:
   918→            if pattern.search(body):
   919→                attachments.append(ImportedAttachment(
   920→                    media_type=media_type,
   921→                    original_path="",
   922→                    file_name="",
   923→                    mime_type="",
   924→                    size_bytes=0,
   925→                ))
   926→        return attachments

  1101→    def _convert_telegram_message(
  1102→        self,
  1103→        raw: dict[str, Any],
  1104→        chat_name: str,
  1105→        chat_type: str,
  1106→        base_dir: Path,
  1107→    ) -> ImportedMessage:
  1108→        """Convert a single Telegram message dict to ImportedMessage.
  1109→
  1110→        Args:
  1111→            raw: Raw message dict from Telegram JSON.
  1112→            chat_name: Name of the conversation.
  1113→            chat_type: Type of chat (personal_chat, group_chat, channel).
  1114→            base_dir: Base directory for resolving attachment paths.
  1115→
  1116→        Returns:
  1117→            ImportedMessage object.
  1118→
  1119→        Raises:
  1120→            ImportValidationError: If the message type is 'service' (system event).
  1121→        """
  1122→        msg_type = raw.get("type", "message")
  1123→        if msg_type == "service":
  1124→            raise ImportValidationError("Service messages are not imported.")
  1125→
  1126→        # Parse ISO 8601 date
  1127→        date_str = raw.get("date", "")
  1128→        timestamp = self._parse_iso_datetime(date_str)
  1129→
  1130→        # Extract text (can be string or list)
  1131→        text = self._extract_telegram_text(raw.get("text", ""))
  1132→
  1133→        sender = raw.get("from", "")
  1134→        sender_id = str(raw.get("from_id", ""))
  1135→        is_outgoing = raw.get("from_id", "") == raw.get("saved_from_id", "")
  1136→
  1137→        # Extract attachments
  1138→        attachments = self._extract_telegram_attachments(raw, base_dir)
  1139→
  1140→        # Extract reply reference
  1141→        reply_to_id = str(raw.get("reply_to_message_id", ""))
  1142→
  1143→        return ImportedMessage(
  1144→            source=ImportSource.TELEGRAM,
  1145→            external_id=str(raw.get("id", uuid.uuid4().hex[:16])),
  1146→            sender_name=sender,
  1147→            sender_id=sender_id,
  1148→            timestamp=timestamp,
  1149→            text=text,
  1150→            attachments=attachments,
  1151→            reply_to_id=reply_to_id,
  1152→            is_outgoing=is_outgoing,
  1153→            chat_name=chat_name,
  1154→            raw_data=raw,
  1155→        )

  1157→    @staticmethod
  1158→    def _parse_iso_datetime(date_str: str) -> float:
  1159→        """Parse an ISO 8601 datetime string to Unix timestamp.
  1160→
  1161→        Args:
  1162→            date_str: ISO 8601 datetime string (e.g., "2023-12-25T10:30:00").
  1163→
  1164→        Returns:
  1165→            Unix timestamp in seconds, or 0.0 if parsing fails.
  1166→        """
  1167→        from datetime import datetime
  1168→
  1169→        if not date_str:
  1170→            return 0.0
  1171→
  1172→        formats = [
  1173→            "%Y-%m-%dT%H:%M:%S",
  1174→            "%Y-%m-%dT%H:%M:%S%z",
  1175→            "%Y-%m-%d %H:%M:%S",
  1176→        ]
  1177→        for fmt in formats:
  1178→            try:
  1179→                dt = datetime.strptime(date_str, fmt)
  1180→                return dt.timestamp()
  1181→            except ValueError:
  1182→                continue
  1183→
  1184→        logger.debug("Failed to parse Telegram datetime: %s", date_str)
  1185→        return 0.0

  1187→    @staticmethod
  1188→    def _extract_telegram_text(text_data: Any) -> str:
  1189→        """Extract plain text from Telegram's mixed text format.
  1190→
  1191→        Telegram text can be:
  1192→            - A simple string: "Hello!"
  1193→            - A list mixing strings and styled objects:
  1194→              ["Hello ", {"type": "bold", "text": "world"}]
  1195→
  1196→        Args:
  1197→            text_data: Raw text field from Telegram JSON.
  1198→
  1199→        Returns:
  1200→            Concatenated plain text string.
  1201→        """
  1202→        if isinstance(text_data, str):
  1203→            return text_data
  1204→
  1205→        if isinstance(text_data, list):
  1206→            parts: list[str] = []
  1207→            for item in text_data:
  1208→                if isinstance(item, str):
  1209→                    parts.append(item)
  1210→                elif isinstance(item, dict) and "text" in item:
  1211→                    parts.append(str(item["text"]))
  1212→            return "".join(parts)
  1213→
  1214→        return ""

  1216→    def _extract_telegram_attachments(
  1217→        self,
  1218→        raw: dict[str, Any],
  1219→        base_dir: Path,
  1220→    ) -> list[ImportedAttachment]:
  1221→        """Extract attachment information from a Telegram message.
  1222→
  1223→        Args:
  1224→            raw: Raw message dict.
  1225→            base_dir: Base directory for resolving file paths.
  1226→
  1227→        Returns:
  1228→            List of ImportedAttachment objects.
  1229→        """
  1230→        attachments: list[ImportedAttachment] = []
  1231→        media_type_str = raw.get("media_type", "")
  1232→        mime_type = raw.get("mime_type", "application/octet-stream")
  1233→
  1234→        # Photo attachment
  1235→        photo = raw.get("photo")
  1236→        if isinstance(photo, dict):
  1237→            uri = photo.get("uri", "")
  1238→            attachments.append(ImportedAttachment(
  1239→                media_type=MessageMediaType.IMAGE,
  1240→                original_path=str(base_dir / uri) if uri else "",
  1241→                file_name=uri.split("/")[-1] if uri else "",
  1242→                mime_type="image/jpeg",
  1243→                size_bytes=0,
  1244→            ))
  1245→
  1246→        # File attachment
  1247→        file_info = raw.get("file")
  1248→        if isinstance(file_info, dict):
  1249→            uri = file_info.get("uri", "")
  1250→            name = file_info.get("name", uri.split("/")[-1] if uri else "")
  1251→            detected_media = self._telegram_media_type(media_type_str, mime_type)
  1252→            attachments.append(ImportedAttachment(
  1253→                media_type=detected_media,
  1254→                original_path=str(base_dir / uri) if uri else "",
  1255→                file_name=name,
  1256→                mime_type=mime_type,
  1257→                size_bytes=0,
  1258→            ))
  1259→
  1260→        # Sticker
  1261→        if media_type_str == "sticker":
  1262→            sticker = raw.get("sticker_emoji", "")
  1263→            attachments.append(ImportedAttachment(
  1264→                media_type=MessageMediaType.STICKER,
  1265→                original_path="",
  1266→                file_name=f"sticker_{sticker}.webp",
  1267→                mime_type="image/webp",
  1268→                size_bytes=0,
  1269→            ))
  1270→
  1271→        # Animation/GIF
  1272→        if media_type_str == "animation":
  1273→            attachments.append(ImportedAttachment(
  1274→                media_type=MessageMediaType.ANIMATION,
  1275→                original_path="",
  1276→                file_name="",
  1277→                mime_type=mime_type or "video/mp4",
  1278→                size_bytes=0,
  1279→            ))
  1280→
  1281→        # Voice message
  1282→        if raw.get("media_type") == "voice_message":
  1283→            attachments.append(ImportedAttachment(
  1284→                media_type=MessageMediaType.VOICE,
  1285→                original_path="",
  1286→                file_name="",
  1287→                mime_type=mime_type or "audio/ogg",
  1288→                size_bytes=0,
  1289→            ))
  1290→
  1291→        return attachments
  1292→
  1293→    @staticmethod
  1294→    def _telegram_media_type(media_type: str, mime_type: str) -> MessageMediaType:
  1295→        """Determine MessageMediaType from Telegram media_type and MIME.
  1296→
  1297→        Args:
  1298→            media_type: Telegram's media_type field value.
  1299→            mime_type: MIME type of the file.
  1300→
  1301→        Returns:
  1302→            Corresponding MessageMediaType value.
  1303→        """
  1304→        mapping: dict[str, MessageMediaType] = {
  1305→            "sticker": MessageMediaType.STICKER,
  1306→            "animation": MessageMediaType.ANIMATION,
  1307→            "voice_message": MessageMediaType.VOICE,
  1308→            "video_message": MessageMediaType.VIDEO,
  1309→        }
  1310→        if media_type in mapping:
  1311→            return mapping[media_type]
  1312→
  1313→        if mime_type.startswith("image/"):
  1314→            return MessageMediaType.IMAGE
  1315→        if mime_type.startswith("video/"):
  1316→            return MessageMediaType.VIDEO
  1317→        if mime_type.startswith("audio/"):
  1318→            return MessageMediaType.AUDIO
  1319→
  1320→        return MessageMediaType.FILE

  1327→class ImportManager:
  1328→    """Orchestrates message imports from external messengers.
  1329→
  1330→    Provides a high-level API for importing conversation history from
  1331→    Signal, WhatsApp, Telegram, and (planned) Matrix. Handles converter
  1332→    selection, validation, batched database insertion, and progress reporting.
  1333→
  1334→    Usage:
  1335→        manager = ImportManager(db_path=Path("~/.presidium/data.db"))
  1336→        result = await manager.import_from_signal(
  1337→            export_path=Path("~/signal-export/"),
  1338→            progress_callback=my_progress_fn,
  1339→        )
  1340→        print(f"Imported {result.imported_messages}/{result.total_messages}")
  1341→
  1342→    Thread safety:
  1343→        ImportManager is designed for single-threaded async usage. If you
  1344→        need to run imports concurrently, use a separate ImportManager
  1345→        instance per task.
  1346→    """
  1347→
  1348→    def __init__(self, db_path: Optional[Path] = None) -> None:
  1349→        self._db_path = db_path
  1350→        self._validators = ValidationPipeline()
  1351→        self._converters: dict[ImportSource, MessageConverter] = {
  1352→            ImportSource.SIGNAL: SignalConverter(),
  1353→            ImportSource.WHATSAPP: WhatsAppConverter(),
  1354→            ImportSource.TELEGRAM: TelegramConverter(),
  1355→        }
  1356→        self._cancelled = False

  1433→    def detect_source(self, export_path: Path) -> Optional[ImportSource]:
  1434→        """Auto-detect the messenger source from the export path.
  1435→
  1436→        Tries each converter's detect_format() method in order.
  1437→
  1438→        Args:
  1439→            export_path: Path to the export file or directory.
  1440→
  1441→        Returns:
  1442→            The detected ImportSource, or None if no converter matches.
  1443→        """
  1444→        for source, converter in self._converters.items():
  1445→            if converter.detect_format(export_path):
  1446→                logger.info(
  1447→                    "Detected export source: %s for path %s",
  1448→                    source.value, export_path,
  1449→                )
  1450→                return source
  1451→        return None

  1458→        """Auto-detect the export format and import.

  1463→

  1467→        Raises:

  1482→
  1483→    async def _run_import(
  1484→        self,
  1485→        source: ImportSource,
  1486→        export_path: Path,
  1487→        progress_callback: Optional[ProgressCallback],
  1488→    ) -> ImportResult:
  1489→        """Execute the full import pipeline for a given source.
  1490→
  1491→        Args:
  1492→            source: Which messenger to import from.
  1493→            export_path: Path to the export data.
  1494→            progress_callback: Optional progress callback.
  1495→
  1496→        Returns:
  1497→            ImportResult with statistics.
  1498→        """
  1499→        self._cancelled = False
  1500→        start_time = time.monotonic()
  1501→
  1502→        converter = self._converters.get(source)
  1503→        if converter is None:
  1504→            raise ImportFormatError(
  1505→                _("No converter available for source: {source}").format(
  1506→                    source=source.value
  1507→                )
  1508→            )
  1509→
  1510→        logger.info(
  1511→            _("Starting import from %s, path: %s"),
  1512→            source.value, export_path,
  1513→        )
  1514→
  1515→        # Phase 1: Parse the export
  1516→        raw_messages = await converter.parse(export_path, progress_callback)
  1517→        total = len(raw_messages)
  1518→
  1519→        if self._cancelled:
  1520→            raise ImportCancelledError(
  1521→                _("Import was cancelled during parsing.")
  1522→            )
  1523→
  1524→        # Phase 2: Validate and sanitize
  1525→        validated_messages: list[ImportedMessage] = []
  1526→        skipped = 0
  1527→        errors: list[str] = []
  1528→
  1529→        for msg in raw_messages:
  1530→            if self._cancelled:
  1531→                raise ImportCancelledError(
  1532→                    _("Import was cancelled during validation.")
  1533→                )
  1534→
  1535→            try:
  1536→                validated = self._validators.validate_message(msg)
  1537→                validated_messages.append(validated)
  1538→            except ImportValidationError as exc:
  1539→                skipped += 1
  1540→                errors.append(str(exc))
  1541→                logger.debug(
  1542→                    "Skipped message %s from %s: %s",
  1543→                    msg.external_id, source.value, exc,
  1544→                )
  1545→
  1546→        # Phase 3: Batch insert into database
  1547→        imported = 0
  1548→        for batch_start in range(0, len(validated_messages), BATCH_SIZE):
  1549→            if self._cancelled:
  1550→                raise ImportCancelledError(
  1551→                    _("Import was cancelled during database insertion.")
  1552→                )
  1553→
  1554→            batch = validated_messages[batch_start:batch_start + BATCH_SIZE]
  1555→            try:
  1556→                inserted = await self._insert_batch(batch)
  1557→                imported += inserted
  1558→            except ImportStorageError as exc:
  1559→                skipped += len(batch)
  1560→                errors.append(str(exc))
  1561→                logger.warning(
  1562→                    "Batch insert failed at offset %d: %s",
  1563→                    batch_start, exc,
  1564→                )
  1565→
  1566→            # Yield control between batches
  1567→            await asyncio.sleep(0)
  1568→
  1569→        duration = time.monotonic() - start_time
  1570→        result = ImportResult(
  1571→            source=source,
  1572→            total_messages=total,
  1573→            imported_messages=imported,
  1574→            skipped_messages=skipped,
  1575→            errors=errors,
  1576→            duration_seconds=round(duration, 3),
  1577→        )
  1578→
  1579→        logger.info(
  1580→            _("Import complete: %d/%d messages imported from %s in %.1fs "
  1581→              "(%d skipped, %d errors)"),
  1582→            imported, total, source.value, duration, skipped, len(errors),
  1583→        )
  1584→
  1585→        return result
  1586→
  1587→    async def _insert_batch(self, messages: list[ImportedMessage]) -> int:
  1588→        """Insert a batch of validated messages into the Presidium database.
  1589→
  1590→        In a full Presidium deployment, this would write to the SQLite
  1591→        database via the db manager module. This implementation performs
  1592→        the validation checks and returns the count of messages that would
  1593→        be inserted.
  1594→
  1595→        Args:
  1596→            messages: List of validated ImportedMessage objects.
  1597→
  1598→        Returns:
  1599→            Number of messages inserted.
  1600→
  1601→        Raises:
  1602→            ImportStorageError: If the database write fails.
  1603→        """
  1604→        if not messages:
  1605→            return 0
  1606→
  1607→        # Validate all messages have required fields for insertion
  1608→        for msg in messages:
  1609→            if not msg.external_id:
  1610→                msg.external_id = uuid.uuid4().hex[:16]
  1611→
  1612→        try:
  1613→            # In production: await db.insert_imported_messages(messages)
  1614→            logger.debug(
  1615→                "Inserting batch of %d messages (source=%s).",
  1616→                len(messages), messages[0].source.value,
  1617→            )
  1618→            return len(messages)
  1619→        except Exception as exc:
  1620→            raise ImportStorageError(
  1621→                _("Database insert failed: {error}").format(error=exc)
  1622→            ) from exc
