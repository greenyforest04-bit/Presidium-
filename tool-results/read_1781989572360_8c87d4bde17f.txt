   337→        bucket = KBucket(k=2)
   338→        n1 = DHTNodeInfo(node_id=os.urandom(16), addr=("1.1.1.1", 1))

   341→        bucket.add(n2)

   360→    @pytest.mark.asyncio
   361→    async def test_full_discovery_roundtrip(self) -> None:
   362→        """Full discovery protocol: register → discover → match."""
   363→        from presidium.core.network.discovery_v2 import ContactDiscoveryV2
   364→
   365→        # Alice registers her contacts
   366→        alice = ContactDiscoveryV2(user_id="alice", key_bits=1024)
   367→        await alice.initialize()
   368→        await alice.register_contacts(["+15551234567", "bob@example.com"])
   369→
   370→        # Bob also registers "bob@example.com"
   371→        bob = ContactDiscoveryV2(user_id="bob", key_bits=1024)
   372→        # Share the same server for the test
   373→        await bob.initialize()
   374→        await bob.register_contacts(["bob@example.com", "carol@example.com"])
   375→
   376→        # Alice discovers using Bob's server
   377→        # (In production the server is shared; here we use Alice's server
   378→        #  and manually register Bob's token there.)
   379→        bob_tokens = bob.client.create_tokens(["bob@example.com"])
   380→        blinded = [t.blinded_token for t in bob_tokens]
   381→        sigs = alice.server.sign_tokens(blinded, "bob")
   382→        completed = bob.client.unblind_tokens(bob_tokens, sigs)
   383→        for token in completed:
   384→            alice.server.register_token(token.unblinded_sig, "bob")
   385→
   386→        # Alice discovers her contacts
   387→        matches = await alice.discover_contacts(["+15551234567", "bob@example.com"])
   388→        assert "bob@example.com" in matches
   389→
   390→        await alice.shutdown()
   391→        await bob.shutdown()

   443→    def test_push_blob_store_and_retrieve(self) -> None:

   445→        from presidium.push.relay_server import PushRelayServer
   446→        server = PushRelayServer()
   447→        server.start()
   448→
   449→        blob_id = hashlib.sha256(os.urandom(32) + struct.pack("!Q", int(time.time()))).digest()[:16]
   450→        encrypted_blob = os.urandom(256)
   451→        hmac_value = hashlib.sha256(blob_id + encrypted_blob).digest()
   452→
   453→        server.store_blob(blob_id, encrypted_blob, hmac_value)
   454→        result = server.get_blob(blob_id)
   455→        assert result == encrypted_blob
   456→
   457→        server.stop()

   468→    def test_blob_expiry(self) -> None:
   469→        """Blobs should be evicted after their expiry time."""
   470→        from presidium.push.relay_server import PushRelayServer
   471→        server = PushRelayServer()
   472→        server.start()
   473→
   474→        blob_id = hashlib.sha256(b"test-expiry").digest()[:16]
   475→        encrypted_blob = os.urandom(128)
   476→        hmac_value = hashlib.sha256(blob_id + encrypted_blob).digest()
   477→
   478→        # Store with immediate expiry (1 second)
   479→        server.store_blob(blob_id, encrypted_blob, hmac_value, ttl_seconds=1)
   480→
   481→        # Should be available now
   482→        assert server.get_blob(blob_id) == encrypted_blob
   483→
   484→        # Wait for expiry
   485→        time.sleep(1.5)
   486→        server._evict_expired()
   487→
   488→        # Should be gone
   489→        assert server.get_blob(blob_id) is None
   490→        server.stop()

   492→    def test_push_rate_limiting(self) -> None:
   493→        """Push relay should rate-limit per IP."""
