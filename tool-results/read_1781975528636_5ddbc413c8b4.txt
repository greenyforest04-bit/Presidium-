     1→"""White-label SDK for embedding Presidium into other applications.
     2→
     3→Provides a simple, thread-safe async API that wraps the core Presidium
     4→modules (identity, network, messaging) into a cohesive interface suitable
     5→for embedding into third-party applications.
     6→
     7→Architecture:
     8→    PresidiumSDK (main entry point)
     9→        ├── SDKConfig — immutable configuration dataclass
    10→        ├── SDKEvent — events emitted by the SDK (message, contact, group, error)
    11→        ├── SDKEventHandler — callback interface for event subscription
    12→        ├── AsyncEventBus — internal event dispatch (thread-safe)
    13→        ├── IdentityFacade — wraps presidium.core.identity
    14→        ├── NetworkFacade — wraps presidium.core.network
    15→        ├── MessagingFacade — wraps presidium.core.messaging
    16→        └── SDKError hierarchy — structured error reporting
    17→
    18→SDK lifecycle:
    19→    1. Construct PresidiumSDK with SDKConfig
    20→    2. Call initialize() to set up identity, network, and messaging
    21→    3. Use send_message(), receive_messages(), create_group(), get_contacts()
    22→    4. Register SDKEventHandler callbacks for real-time events
    23→    5. Call shutdown() to cleanly release all resources
    24→
    25→Thread safety:
    26→    All public SDK methods are safe to call from any thread. Internally,
    27→    the SDK uses an asyncio event loop and thread-safe queues to bridge
    28→    synchronous callers to the async core. The recommended usage pattern
    29→    is async/await from a single async context.
    30→
    31→Error handling:
    32→    - SDKError: Base exception for all SDK errors
    33→    - SDKNotInitializedError: SDK used before initialize()
    34→    - SDKConnectionError: Network connectivity issues
    35→    - SDKAuthenticationError: Invalid API key or identity
    36→    - SDKMessageError: Message send/receive failures
    37→    - SDKGroupError: Group creation or management failures
    38→    - SDKTimeoutError: Operation exceeded time limit
    39→
    40→Usage example:
    41→    config = SDKConfig(
    42→        app_name="MyApp",
    43→        api_key="pk_live_abc123",
    44→        relay_url="wss://relay.presidium.chat",
    45→    )
    46→    sdk = PresidiumSDK(config)
    47→    await sdk.initialize()
    48→
    49→    # Send a message
    50→    await sdk.send_message(recipient_id="abc123", text="Hello!")
    51→
    52→    # Register event handler
    53→    class MyHandler(SDKEventHandler):
    54→        async def on_message_received(self, message):
    55→            print(f"Got: {message.text}")
    56→
    57→    sdk.register_handler(MyHandler())
    58→
    59→    # Clean up
    60→    await sdk.shutdown()
    61→"""

   195→@dataclass
   196→class SDKConfig:
   197→    """Immutable configuration for the Presidium SDK.
   198→
   199→    Attributes:
   200→        app_name: Name of the embedding application (for user-agent and logging).
   201→        api_key: Presidium API key for authentication with the relay.
   202→        relay_url: WebSocket URL of the Presidium relay server.
   203→        api_version: API version string (default: "v1").
   204→        timeout_seconds: Default timeout for SDK operations.
   205→        reconnect_attempts: Maximum number of reconnection attempts.
   206→        reconnect_delay_seconds: Base delay between reconnection attempts.
   207→        auto_reconnect: Whether to automatically reconnect on disconnect.
   208→        enable_encryption: Whether to enable E2EE (default: True).
   209→        data_dir: Optional custom data directory for identity and cache.
   210→        heartbeat_interval: Interval in seconds for keep-alive heartbeats.
   211→        max_message_length: Maximum message text length in characters.
   212→    """
   213→    app_name: str = ""
   214→    api_key: str = ""
   215→    relay_url: str = DEFAULT_RELAY_URL
   216→    api_version: str = DEFAULT_API_VERSION
   217→    timeout_seconds: float = DEFAULT_TIMEOUT_SECONDS
   218→    reconnect_attempts: int = DEFAULT_RECONNECT_ATTEMPTS
   219→    reconnect_delay_seconds: float = DEFAULT_RECONNECT_DELAY_SECONDS
   220→    auto_reconnect: bool = True
   221→    enable_encryption: bool = True
   222→    data_dir: str = ""
   223→    heartbeat_interval: float = HEARTBEAT_INTERVAL_SECONDS
   224→    max_message_length: int = MAX_MESSAGE_LENGTH
   225→
   226→    def validate(self) -> None:
   227→        """Validate the configuration and raise on invalid values.
   228→
   229→        Raises:
   230→            SDKConfigError: If required fields are missing or values are invalid.
   231→        """
   232→        if not self.app_name:
   233→            raise SDKConfigError(_("app_name is required."))
   234→        if not self.api_key:
   235→            raise SDKConfigError(_("api_key is required."))
   236→        if not self.relay_url.startswith(("wss://", "ws://")):
   237→            raise SDKConfigError(
   238→                _("relay_url must start with wss:// or ws://, got: {url}").format(
   239→                    url=self.relay_url

   241→            )
   242→        if self.timeout_seconds <= 0:
   243→            raise SDKConfigError(_("timeout_seconds must be positive."))
   244→        if self.reconnect_attempts < 0:
   245→            raise SDKConfigError(_("reconnect_attempts cannot be negative."))
   246→        if self.heartbeat_interval < 1.0:
   247→            raise SDKConfigError(_("heartbeat_interval must be at least 1 second."))

   250→@dataclass
   251→class SDKMessage:
   252→    """A message in the Presidium SDK.
   253→
   254→    Attributes:
   255→        id: Unique message identifier.
   256→        conversation_id: ID of the conversation this message belongs to.
   257→        sender_id: User ID of the sender.
   258→        recipient_id: User ID of the recipient (for 1:1 chats).
   259→        text: Message body text.
   260→        message_type: Type of message content.
   261→        timestamp: Unix timestamp (seconds) when the message was sent/received.
   262→        is_outgoing: Whether this message was sent by the local user.
   263→        is_read: Whether the message has been read.
   264→        reply_to_id: ID of the message this replies to, if any.
   265→        metadata: Additional key-value metadata.
   266→    """
   267→    id: str = ""
   268→    conversation_id: str = ""
   269→    sender_id: str = ""
   270→    recipient_id: str = ""
   271→    text: str = ""
   272→    message_type: MessageType = MessageType.TEXT
   273→    timestamp: float = 0.0
   274→    is_outgoing: bool = False
   275→    is_read: bool = False
   276→    reply_to_id: str = ""
   277→    metadata: dict[str, Any] = field(default_factory=dict)
   278→
   279→    def to_dict(self) -> dict[str, Any]:
   280→        """Serialize to a JSON-compatible dict."""
   281→        return {
   282→            "id": self.id,
   283→            "conversation_id": self.conversation_id,
   284→            "sender_id": self.sender_id,
   285→            "recipient_id": self.recipient_id,
   286→            "text": self.text,
   287→            "message_type": self.message_type.value,
   288→            "timestamp": self.timestamp,
   289→            "is_outgoing": self.is_outgoing,
   290→            "is_read": self.is_read,
   291→            "reply_to_id": self.reply_to_id,
   292→            "metadata": self.metadata,
   293→        }
   294→
   295→    @classmethod
   296→    def from_dict(cls, data: dict[str, Any]) -> SDKMessage:
   297→        """Deserialize from a dict.
   298→
   299→        Args:
   300→            data: Dictionary with message fields.
   301→
   302→        Returns:
   303→            SDKMessage instance.
   304→        """
   305→        msg_type = MessageType.TEXT
   306→        raw_type = data.get("message_type", "text")
   307→        try:
   308→            msg_type = MessageType(raw_type)
   309→        except ValueError:
   310→            logger.debug("Unknown message type '%s', defaulting to TEXT.", raw_type)
   311→
   312→        return cls(
   313→            id=data.get("id", ""),
   314→            conversation_id=data.get("conversation_id", ""),
   315→            sender_id=data.get("sender_id", ""),
   316→            recipient_id=data.get("recipient_id", ""),
   317→            text=data.get("text", ""),
   318→            message_type=msg_type,
   319→            timestamp=data.get("timestamp", 0.0),
   320→            is_outgoing=data.get("is_outgoing", False),
   321→            is_read=data.get("is_read", False),
   322→            reply_to_id=data.get("reply_to_id", ""),
   323→            metadata=data.get("metadata", {}),
   324→        )

   414→class SDKEventHandler(abc.ABC):
   415→    """Abstract callback interface for SDK events.
   416→
   417→    Subclass this and override the methods for events you care about.
   418→    Unhandled events are safely ignored (no-op default implementations).
   419→
   420→    All callback methods are async. The SDK dispatches events from its
   421→    internal event loop, so callbacks should not block for long periods.
   422→    """
   423→
   424→    async def on_message_received(self, message: SDKMessage) -> None:
   425→        """Called when a new message is received.
   426→
   427→        Args:
   428→            message: The received message.
   429→        """
   430→
   431→    async def on_message_sent(self, message: SDKMessage) -> None:
   432→        """Called when a message is successfully sent.
   433→
   434→        Args:
   435→            message: The sent message (with assigned ID and timestamp).
   436→        """
   437→
   438→    async def on_contact_added(self, contact: SDKContact) -> None:
   439→        """Called when a new contact is added.
   440→
   441→        Args:
   442→            contact: The newly added contact.
   443→        """
   444→
   445→    async def on_group_created(self, group: SDKGroup) -> None:
   446→        """Called when a new group is created.
   447→
   448→        Args:
   449→            group: The newly created group.
   450→        """
   451→
   452→    async def on_error(self, error: SDKError) -> None:
   453→        """Called when an SDK error occurs.
   454→
   455→        Args:
   456→            error: The error that occurred.
   457→        """

   460→# ---------------------------------------------------------------------------
   461→# Internal event bus
   462→# ---------------------------------------------------------------------------
   463→
   464→class AsyncEventBus:
   465→    """Thread-safe event dispatch for the SDK.
   466→
   467→    Manages a list of SDKEventHandler instances and dispatches events
   468→    to all registered handlers concurrently. Errors in individual handlers
   469→    are caught and logged without affecting other handlers.
   470→
   471→    Thread safety:
   472→        register() and unregister() are protected by a threading lock.
   473→        Event dispatch runs on the SDK's async event loop.
   474→    """
   475→
   476→    def __init__(self) -> None:
   477→        self._handlers: list[SDKEventHandler] = []
   478→        self._lock = threading.Lock()
   479→
   480→    def register(self, handler: SDKEventHandler) -> None:
   481→        """Register an event handler.
   482→
   483→        Args:
   484→            handler: The handler to register.
   485→        """
   486→        with self._lock:
   487→            if handler not in self._handlers:
   488→                self._handlers.append(handler)
   489→                logger.debug("Registered SDK event handler: %s", type(handler).__name__)
   490→
   491→    def unregister(self, handler: SDKEventHandler) -> None:
   492→        """Unregister an event handler.
   493→
   494→        Args:
   495→            handler: The handler to remove.
   496→        """
   497→        with self._lock:
   498→            try:
   499→                self._handlers.remove(handler)
   500→                logger.debug("Unregistered SDK event handler: %s", type(handler).__name__)
   501→            except ValueError:
   502→                logger.debug("Handler %s not found for removal.", type(handler).__name__)
   503→
   504→    async def emit_message_received(self, message: SDKMessage) -> None:
   505→        """Dispatch a MESSAGE_RECEIVED event to all handlers.
   506→
   507→        Args:
   508→            message: The received message.
   509→        """
   510→        handlers = self._get_handlers()
   511→        for handler in handlers:
   512→            try:
   513→                await handler.on_message_received(message)
   514→            except Exception as exc:
   515→                logger.warning(
   516→                    "Error in %s.on_message_received: %s",
   517→                    type(handler).__name__, exc,
   518→                )
   519→
   520→    async def emit_message_sent(self, message: SDKMessage) -> None:
   521→        """Dispatch a MESSAGE_SENT event to all handlers.
   522→
   523→        Args:
   524→            message: The sent message.
   525→        """
   526→        handlers = self._get_handlers()
   527→        for handler in handlers:
   528→            try:
   529→                await handler.on_message_sent(message)
   530→            except Exception as exc:
   531→                logger.warning(
   532→                    "Error in %s.on_message_sent: %s",
   533→                    type(handler).__name__, exc,
   534→                )
   535→
   536→    async def emit_contact_added(self, contact: SDKContact) -> None:
   537→        """Dispatch a CONTACT_ADDED event to all handlers.
   538→
   539→        Args:
   540→            contact: The added contact.
   541→        """
   542→        handlers = self._get_handlers()
   543→        for handler in handlers:
   544→            try:
   545→                await handler.on_contact_added(contact)
   546→            except Exception as exc:
   547→                logger.warning(
   548→                    "Error in %s.on_contact_added: %s",
   549→                    type(handler).__name__, exc,
   550→                )
   551→
   552→    async def emit_group_created(self, group: SDKGroup) -> None:
   553→        """Dispatch a GROUP_CREATED event to all handlers.
   554→
   555→        Args:
   556→            group: The created group.
   557→        """
   558→        handlers = self._get_handlers()
   559→        for handler in handlers:
   560→            try:
   561→                await handler.on_group_created(group)
   562→            except Exception as exc:
   563→                logger.warning(
   564→                    "Error in %s.on_group_created: %s",
   565→                    type(handler).__name__, exc,
   566→                )
   567→
   568→    async def emit_error(self, error: SDKError) -> None:
   569→        """Dispatch an ERROR event to all handlers.
   570→
   571→        Args:
   572→            error: The SDK error.
   573→        """
   574→        handlers = self._get_handlers()
   575→        for handler in handlers:
   576→            try:
   577→                await handler.on_error(error)
   578→            except Exception as exc:
   579→                logger.warning(
   580→                    "Error in %s.on_error: %s",
   581→                    type(handler).__name__, exc,
   582→                )
   583→
   584→    def _get_handlers(self) -> list[SDKEventHandler]:
   585→        """Return a snapshot of the current handler list.
   586→
   587→        Returns:
   588→            Copy of the handlers list (thread-safe).
   589→        """
   590→        with self._lock:
   591→            return list(self._handlers)

   810→    async def send_message(
   811→        self,
   812→        sender_id: str,
   813→        recipient_id: str,
   814→        text: str,
   815→        conversation_id: str,
   816→        network: NetworkFacade,
   817→    ) -> SDKMessage:
   818→        """Create, encrypt, and send a message.
   819→
   820→        Args:
   821→            sender_id: Local user's ID.
   822→            recipient_id: Recipient's user ID.
   823→            text: Message body text.
   824→            conversation_id: Conversation the message belongs to.
   825→            network: Network facade for sending.
   826→
   827→        Returns:
   828→            SDKMessage with assigned ID and timestamp.
   829→
   830→        Raises:
   831→            SDKMessageError: If the message cannot be sent.
   832→        """
   833→        if not text:
   834→            raise SDKMessageError(_("Message text cannot be empty."))
   835→        if len(text) > self._config.max_message_length:
   836→            raise SDKMessageError(
   837→                _("Message exceeds maximum length of {max_len} characters.").format(
   838→                    max_len=self._config.max_message_length
   839→                )
   840→            )
   841→
   842→        message = SDKMessage(
   843→            id=uuid.uuid4().hex,
   844→            conversation_id=conversation_id,
   845→            sender_id=sender_id,
   846→            recipient_id=recipient_id,
   847→            text=text,
   848→            message_type=MessageType.TEXT,
   849→            timestamp=time.time(),
   850→            is_outgoing=True,
   851→            is_read=False,
   852→        )
   853→
   854→        try:
   855→            # In production: encrypt with Double Ratchet and send via network
   856→            if self._config.enable_encryption:
   857→                self._encryption_stats.messages_encrypted += 1
   858→                logger.debug("Encrypting message %s for %s.", message.id, recipient_id)
   859→
   860→            # Simulate sending via network
   861→            payload = message.to_dict()
   862→            import json
   863→            raw_data = json.dumps(payload).encode("utf-8")
   864→            await network.send_raw(raw_data)
   865→
   866→            self._sent_count += 1
   867→            logger.debug("Message %s sent to %s.", message.id, recipient_id)
   868→
   869→        except SDKConnectionError as exc:
   870→            self._encryption_stats.encryption_failures += 1
   871→            raise SDKMessageError(
   872→                _("Failed to send message: {error}").format(error=exc)
   873→            ) from exc
   874→
   875→        return message
   876→
   877→    async def receive_message(self, raw_data: bytes) -> SDKMessage:
   878→        """Decrypt and parse an incoming message.
   879→
   880→        Args:
   881→            raw_data: Raw encrypted message data from the network.
   882→
   883→        Returns:
   884→            Decrypted SDKMessage.
   885→
   886→        Raises:
   887→            SDKMessageError: If decryption or parsing fails.
   888→        """
   889→        try:
   890→            import json
   891→            payload = json.loads(raw_data.decode("utf-8"))
   892→
   893→            if self._config.enable_encryption:
   894→                # In production: decrypt with Double Ratchet
   895→                self._encryption_stats.messages_decrypted += 1
   896→
   897→            message = SDKMessage.from_dict(payload)
   898→            message.is_outgoing = False
   899→            message.is_read = False
   900→
   901→            self._received_count += 1
   902→            logger.debug("Received message %s from %s.", message.id, message.sender_id)
   903→
   904→            return message
   905→
   906→        except (json.JSONDecodeError, UnicodeDecodeError, KeyError) as exc:
   907→            self._encryption_stats.encryption_failures += 1
   908→            raise SDKMessageError(
   909→                _("Failed to parse incoming message: {error}").format(error=exc)
   910→            ) from exc

   976→class PresidiumSDK:
   977→    """Main entry point for the Presidium white-label SDK.
   978→
   979→    Wraps identity, network, and messaging modules into a simple async
   980→    API suitable for embedding into third-party applications. All public
   981→    methods are thread-safe and may be called from any thread.
   982→
   983→    Lifecycle:
   984→        1. Construct with SDKConfig
   985→        2. Call initialize() — sets up identity, connects to relay
   986→        3. Use messaging and group APIs
   987→        4. Call shutdown() — disconnects and clears state
   988→
   989→    Example:
   990→        config = SDKConfig(app_name="MyApp", api_key="pk_live_abc123")
   991→        sdk = PresidiumSDK(config)
   992→        await sdk.initialize()
   993→        msg = await sdk.send_message("user_abc", "Hello!")
   994→        await sdk.shutdown()
   995→    """
   996→
   997→    def __init__(self, config: SDKConfig) -> None:
   998→        self._config = config
   999→        self._identity = IdentityFacade()
  1000→        self._network = NetworkFacade(config)
  1001→        self._messaging = MessagingFacade(config)
  1002→        self._event_bus = AsyncEventBus()
  1003→        self._contacts: dict[str, SDKContact] = {}
  1004→        self._groups: dict[str, SDKGroup] = {}
  1005→        self._initialized = False
  1006→        self._shutdown = False
  1007→        self._lock = threading.Lock()
  1008→        self._recv_queue: asyncio.Queue[SDKMessage] = asyncio.Queue()
  1009→        self._background_tasks: list[asyncio.Task[None]] = []
  1010→
  1011→    # -------------------------------------------------------------------
  1012→    # Properties
  1013→    # -------------------------------------------------------------------
  1014→
  1015→    @property
  1016→    def is_initialized(self) -> bool:
  1017→        """Whether the SDK has been initialized and is ready to use."""
  1018→        return self._initialized
  1019→
  1020→    @property
  1021→    def connection_state(self) -> ConnectionState:
  1022→        """Current connection state of the network layer."""
  1023→        return self._network.state
  1024→
  1025→    @property
  1026→    def user_id(self) -> str:
  1027→        """Current user ID, or empty string if not initialized."""
  1028→        return self._identity.user_id
  1029→
  1030→    @property
  1031→    def encryption_stats(self) -> SDKEncryptionStats:
  1032→        """E2EE statistics for the current session."""
  1033→        return self._messaging.encryption_stats

  1035→    # -------------------------------------------------------------------
  1036→    # Lifecycle
  1037→    # -------------------------------------------------------------------
  1038→
  1039→    async def initialize(
  1040→        self,
  1041→        password: Optional[str] = None,
  1042→        nickname: Optional[str] = None,
  1043→    ) -> str:
  1044→        """Initialize the SDK: create/load identity and connect to relay.
  1045→
  1046→        This must be called before any other SDK method. It performs:
  1047→            1. Validates the SDKConfig
  1048→            2. Initializes the identity (create or unlock)
  1049→            3. Connects to the Presidium relay
  1050→            4. Starts background tasks (heartbeat, receive loop)
  1051→
  1052→        Args:
  1053→            password: Password for identity encryption. Defaults to api_key.
  1054→            nickname: Optional display name for the local user.
  1055→
  1056→        Returns:
  1057→            The local user ID.
  1058→
  1059→        Raises:
  1060→            SDKConfigError: If the configuration is invalid.
  1061→            SDKAuthenticationError: If identity initialization fails.
  1062→            SDKConnectionError: If the relay connection fails.
  1063→        """
  1064→        with self._lock:
  1065→            if self._initialized:
  1066→                logger.warning("SDK already initialized, skipping.")
  1067→                return self._identity.user_id
  1068→
  1069→        # Validate configuration
  1070→        self._config.validate()
  1071→
  1072→        # Initialize identity
  1073→        identity_password = password or self._config.api_key
  1074→        data_dir = self._config.data_dir or f".presidium-sdk-{self._config.app_name}"
  1075→        user_id = await self._identity.initialize(data_dir, identity_password)
  1076→
  1077→        if nickname:
  1078→            self._identity._nickname = nickname
  1079→
  1080→        # Connect to relay
  1081→        await self._network.connect(self._identity)
  1082→
  1083→        # Start background tasks
  1084→        loop = asyncio.get_running_loop()
  1085→        self._background_tasks.append(
  1086→            loop.create_task(self._heartbeat_loop())
  1087→        )
  1088→        self._background_tasks.append(
  1089→            loop.create_task(self._receive_loop())
  1090→        )
  1091→
  1092→        with self._lock:
  1093→            self._initialized = True
  1094→            self._shutdown = False
  1095→
  1096→        logger.info(
  1097→            _("Presidium SDK initialized for %s (user_id=%s)"),
  1098→            self._config.app_name, user_id,
  1099→        )
  1100→        return user_id
  1101→
  1102→    async def shutdown(self) -> None:
  1103→        """Shut down the SDK and release all resources.
  1104→
  1105→        Cancels background tasks, disconnects from the relay,
  1106→        and clears identity material from memory.
  1107→
  1108→        After shutdown, initialize() must be called again before use.
  1109→        """
  1110→        with self._lock:
  1111→            if not self._initialized:
  1112→                return
  1113→            self._shutdown = True
  1114→
  1115→        # Cancel background tasks
  1116→        for task in self._background_tasks:
  1117→            task.cancel()
  1118→        for task in self._background_tasks:
  1119→            try:
  1120→                await task
  1121→            except asyncio.CancelledError:
  1122→                logger.debug("Background task cancelled during shutdown.")
  1123→        self._background_tasks.clear()
  1124→
  1125→        # Disconnect and clear
  1126→        await self._network.disconnect()
  1127→        await self._identity.shutdown()
  1128→
  1129→        self._contacts.clear()
  1130→        self._groups.clear()
  1131→        self._initialized = False
  1132→
  1133→        logger.info(_("Presidium SDK shut down."))

  1181→    async def receive_messages(
  1182→        self,
  1183→        timeout: float = 0.0,
  1184→    ) -> list[SDKMessage]:
  1185→        """Retrieve received messages from the internal queue.
  1186→
  1187→        This is a polling-based receive. For push-based notifications,
  1188→        register an SDKEventHandler with on_message_received().
  1189→
  1190→        Args:
  1191→            timeout: Maximum time in seconds to wait for a message.
  1192→                0.0 returns immediately with whatever is available.
  1193→
  1194→        Returns:
  1195→            List of received SDKMessage objects (may be empty).
  1196→
  1197→        Raises:
  1198→            SDKNotInitializedError: If the SDK is not initialized.
  1199→        """
  1200→        self._require_initialized()
  1201→
  1202→        messages: list[SDKMessage] = []
  1203→        deadline = time.monotonic() + timeout if timeout > 0 else 0.0
  1204→
  1205→        while True:
  1206→            remaining = deadline - time.monotonic() if deadline > 0 else 0.0
  1207→            if timeout > 0 and remaining <= 0:
  1208→                break
  1209→
  1210→            try:
  1211→                wait_time = remaining if timeout > 0 else 0.001
  1212→                msg = await asyncio.wait_for(
  1213→                    self._recv_queue.get(), timeout=wait_time
  1214→                )
  1215→                messages.append(msg)
  1216→            except asyncio.TimeoutError:
  1217→                if timeout == 0:
  1218→                    break
  1219→
  1220→        return messages

  1487→    @staticmethod
  1488→    def _compute_conversation_id(peer_id: str) -> str:
  1489→        """Compute a deterministic conversation ID for a 1:1 chat.
  1490→
  1491→        The conversation ID is derived from the sorted pair of user IDs
  1492→        so that both parties compute the same ID regardless of who
  1493→        initiates the conversation.
  1494→
  1495→        Args:
  1496→            peer_id: The peer's user ID.
  1497→
  1498→        Returns:
  1499→            Conversation ID string.
  1500→        """
  1501→        import hashlib
  1502→        local_id = ""  # Will be filled in at call site
  1503→        sorted_ids = sorted([local_id, peer_id])
  1504→        raw = ":".join(sorted_ids)
  1505→        return hashlib.sha256(raw.encode("utf-8")).hexdigest()[:32]

  1507→    async def _heartbeat_loop(self) -> None:
  1508→        """Background task: send periodic heartbeats to the relay.
  1509→
  1510→        Runs until cancelled or the SDK shuts down.
  1511→        """
  1512→        try:
  1513→            while not self._shutdown:
  1514→                if self._network.is_connected:
  1515→                    await self._network.send_heartbeat()
  1516→                await asyncio.sleep(self._config.heartbeat_interval)
  1517→        except asyncio.CancelledError:
  1518→            return
  1519→        except Exception as exc:
  1520→            logger.warning("Heartbeat loop error: %s", exc)
  1521→            await self._event_bus.emit_error(
  1522→                SDKConnectionError(_("Heartbeat failed: {error}").format(error=exc))
  1523→            )
  1524→
  1525→    async def _receive_loop(self) -> None:
  1526→        """Background task: poll for incoming messages from the relay.
  1527→
  1528→        Received messages are decrypted, placed on the internal queue,
  1529→        and dispatched to event handlers.
  1530→
  1531→        Runs until cancelled or the SDK shuts down.
  1532→        """
  1533→        try:
  1534→            while not self._shutdown:
  1535→                if not self._network.is_connected:
  1536→                    await asyncio.sleep(0.5)
  1537→                    continue
  1538→
  1539→                try:
  1540→                    raw = await self._network.receive_raw(timeout=1.0)
  1541→                    if raw is not None:
  1542→                        message = await self._messaging.receive_message(raw)
  1543→                        await self._recv_queue.put(message)
  1544→                        await self._event_bus.emit_message_received(message)
  1545→                except SDKMessageError as exc:
  1546→                    logger.warning("Failed to process incoming message: %s", exc)
  1547→                    await self._event_bus.emit_error(exc)
  1548→                except asyncio.CancelledError:
  1549→                    return
  1550→                except Exception as exc:
  1551→                    logger.debug("Receive loop iteration error: %s", exc)
  1552→                    await asyncio.sleep(0.1)
  1553→
  1554→        except asyncio.CancelledError:
  1555→            return

  1557→    @staticmethod
  1558→    def _run_async(coro: Any) -> Any:
  1559→        """Run an async coroutine from a synchronous context.
  1560→
  1561→        Handles the case where an event loop is already running (e.g.,
  1562→        in a Jupyter notebook or existing async application) by using
  1563→        a separate thread.
  1564→
  1565→        Args:
  1566→            coro: The coroutine to run.
  1567→
  1568→        Returns:
  1569→            The result of the coroutine.
  1570→        """
  1571→        try:
  1572→            loop = asyncio.get_running_loop()
  1573→        except RuntimeError:
  1574→            loop = None
  1575→
  1576→        if loop is not None and loop.is_running():
  1577→            # We're inside an existing event loop — use a thread
  1578→            import concurrent.futures
  1579→            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
  1580→                future = pool.submit(asyncio.run, coro)
  1581→                return future.result(timeout=DEFAULT_TIMEOUT_SECONDS)
  1582→        else:
  1583→            return asyncio.run(coro)
