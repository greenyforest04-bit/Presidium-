     1→"""Group Voice/Video Calls for Presidium — P2P SFU architecture.
     2→
     3→Implements group calls using a hybrid mesh/SFU topology:
     4→    - Mesh (all-to-all): 2-3 participants — lowest latency, no single point
     5→    - SFU (host forwards): 4-16 participants — host acts as Selective
     6→      Forwarding Unit, receiving all streams and selectively forwarding
     7→      to each participant based on bandwidth and subscription.
     8→
     9→SFU model (P2P, no central server):
    10→    The call initiator becomes the "host" and runs a lightweight SFU:
    11→    1. All participants send their media stream to the host
    12→    2. Host selectively forwards streams to each participant
    13→    3. Host does NOT mix/encode — only forwards encoded packets
    14→    4. Each participant receives only the streams they subscribe to
    15→
    16→Simulcast:
    17→    Each sender encodes 3 quality layers:
    18→    - Low:  320x180 @ 15fps, 150 kbps
    19→    - Mid:  640x360 @ 24fps, 400 kbps
    20→    - High: 1280x720 @ 30fps, 800 kbps
    21→    The SFU switches layers based on receiver bandwidth.
    22→
    23→Signaling:
    24→    All signaling uses the existing P2P transport with new
    25→    GROUP_CALL_* message types for offer/answer/ICE/candidate exchange.
    26→
    27→Limits:
    28→    - Maximum 16 participants per call
    29→    - Video: up to 720p per participant (bandwidth constrained)
    30→    - Audio: OPUS codec, 48kHz, mono
    31→    - Screen share: one participant at a time
    32→
    33→Call state machine:
    34→    IDLE → RINGING → JOINING → CONNECTED → LEAVING → IDLE
    35→"""

    60→# Simulcast layers

    75→# ---------------------------------------------------------------------------
    76→# Call state machine
    77→# ---------------------------------------------------------------------------
    78→
    79→class CallState(enum.IntEnum):
    80→    """State of a group call."""
    81→    IDLE = 0
    82→    RINGING = 1
    83→    JOINING = 2
    84→    CONNECTED = 3
    85→    RECONNECTING = 4
    86→    LEAVING = 5
    87→    ENDED = 6
    88→
    89→
    90→class CallTopology(enum.IntEnum):
    91→    """Call network topology."""
    92→    MESH = 0    # All-to-all (2-3 participants)
    93→    SFU = 1     # Host-based forwarding (4-16 participants)
    94→
    95→
    96→class MediaState(enum.IntEnum):
    97→    """Participant media state."""
    98→    OFF = 0
    99→    ON = 1
   100→    MUTED = 2      # Audio muted
   101→    PAUSED = 3     # Video paused

   108→@dataclass
   109→class CallParticipant:
   110→    """A participant in a group call."""
   111→    user_id: str = ""
   112→    display_name: str = ""
   113→    is_host: bool = False
   114→    audio_state: MediaState = MediaState.OFF
   115→    video_state: MediaState = MediaState.OFF
   116→    screen_share: bool = False
   117→    is_raising_hand: bool = False
   118→    joined_at: float = 0.0
   119→    last_seen: float = 0.0
   120→    connection_quality: float = 1.0  # 0.0-1.0
   121→
   122→    # WebRTC peer connection (set at runtime)
   123→    peer_connection: Any = None  # aiortc.RTCPeerConnection
   124→
   125→    # Simulcast subscription (which quality layer this participant receives)
   126→    subscribed_quality: str = "mid"  # "low" | "mid" | "high"
   127→
   128→    @property
   129→    def is_active(self) -> bool:
   130→        """Check if the participant is still connected."""
   131→        return self.audio_state != MediaState.OFF or self.video_state != MediaState.OFF

   134→# ---------------------------------------------------------------------------
   135→# Call session
   136→# ---------------------------------------------------------------------------
   137→
   138→@dataclass
   139→class GroupCallSession:
   140→    """Represents an active group call session."""
   141→    call_id: str = field(default_factory=lambda: uuid.uuid4().hex[:16])
   142→    group_id: str = ""          # Associated group chat ID
   143→    host_user_id: str = ""      # Call initiator / SFU host
   144→    topology: CallTopology = CallTopology.MESH
   145→    state: CallState = CallState.IDLE
   146→    participants: dict[str, CallParticipant] = field(default_factory=dict)
   147→    created_at: float = field(default_factory=time.time)
   148→    started_at: float = 0.0
   149→    ended_at: float = 0.0
   150→    is_recording: bool = False
   151→    is_screen_sharing: bool = False
   152→    screen_sharer_id: str = ""
   153→
   154→    @property
   155→    def participant_count(self) -> int:
   156→        return len(self.participants)
   157→
   158→    @property
   159→    def is_full(self) -> bool:
   160→        return self.participant_count >= MAX_PARTICIPANTS
   161→
   162→    @property
   163→    def is_host(self) -> bool:
   164→        """True if the local user is the call host."""
   165→        return False  # Set by GroupCallManager
   166→
   167→    def add_participant(self, participant: CallParticipant) -> None:
   168→        """Add a participant to the call."""
   169→        if self.is_full:
   170→            raise GroupCallError("Call is full (max 16 participants).")
   171→        self.participants[participant.user_id] = participant
   172→
   173→    def remove_participant(self, user_id: str) -> Optional[CallParticipant]:
   174→        """Remove a participant from the call."""
   175→        return self.participants.pop(user_id, None)
   176→
   177→    def get_participant(self, user_id: str) -> Optional[CallParticipant]:
   178→        """Get a participant by user ID."""
   179→        return self.participants.get(user_id)
   180→
   181→    def get_active_speakers(self) -> list[CallParticipant]:
   182→        """Get participants with audio enabled (potential speakers)."""
   183→        return [
   184→            p for p in self.participants.values()
   185→            if p.audio_state == MediaState.ON
   186→        ]
   187→
   188→    def determine_topology(self) -> CallTopology:
   189→        """Determine the appropriate topology based on participant count."""
   190→        if self.participant_count <= MESH_THRESHOLD:
   191→            return CallTopology.MESH
   192→        return CallTopology.SFU

   400→    # Call lifecycle
   401→    # -------------------------------------------------------------------
   402→
   403→    async def start_call(
   404→        self,
   405→        group_id: str,
   406→        participants: list[str],
   407→        with_video: bool = False,
   408→        with_screen_share: bool = False,
   409→    ) -> GroupCallSession:
   410→        """Initiate a group call.
   411→
   412→        Args:
   413→            group_id: The group chat ID this call belongs to.
   414→            participants: List of user_ids to invite.
   415→            with_video: Whether to start with video enabled.
   416→            with_screen_share: Whether to start with screen sharing.
   417→
   418→        Returns:
   419→            The created GroupCallSession.
   420→        """
   421→        if len(participants) + 1 > MAX_PARTICIPANTS:
   422→            raise GroupCallError(
   423→                f"Too many participants: {len(participants) + 1} > {MAX_PARTICIPANTS}."
   424→            )
   425→
   426→        session = GroupCallSession(
   427→            group_id=group_id,
   428→            host_user_id=self._local_user_id,
   429→            state=CallState.RINGING,
   430→        )
   431→
   432→        # Determine topology
   433→        session.topology = session.determine_topology()
   434→
   435→        # Add self as host participant
   436→        host = CallParticipant(
   437→            user_id=self._local_user_id,
   438→            is_host=True,
   439→            audio_state=MediaState.ON,
   440→            video_state=MediaState.ON if with_video else MediaState.OFF,
   441→            screen_share=with_screen_share,
   442→            joined_at=time.time(),
   443→        )
   444→        session.add_participant(host)
   445→
   446→        # Create WebRTC offer (aiortc)
   447→        sdp = await self._create_webrtc_offer(session)
   448→
   449→        # Send invites to all participants
   450→        for peer_id in participants:
   451→            signal = GroupCallSignal.create_offer(
   452→                call_id=session.call_id,
   453→                group_id=group_id,
   454→                host_user_id=self._local_user_id,
   455→                sdp=sdp,
   456→                media_types=["audio"] + (["video"] if with_video else []),
   457→            )
   458→            await self._send_signal(peer_id, signal)
   459→
   460→        self._active_calls[session.call_id] = session
   461→
   462→        if self._on_call_state_changed:
   463→            await self._on_call_state_changed(session)
   464→
   465→        logger.info(
   466→            "Group call started: %s (%s, %d invited)",
   467→            session.call_id, session.topology.name, len(participants),
   468→        )
   469→        return session
   470→
   471→    async def join_call(
   472→        self,
   473→        call_id: str,
   474→        group_id: str,
   475→        host_user_id: str,
   476→        sdp_offer: str,
   477→    ) -> GroupCallSession:
   478→        """Join an existing group call.
   479→
   480→        Args:
   481→            call_id: The call ID from the offer.
   482→            group_id: The group chat ID.
   483→            host_user_id: The call host's user ID.
   484→            sdp_offer: The WebRTC SDP offer.
   485→
   486→        Returns:
   487→            The joined GroupCallSession.
   488→        """
   489→        if call_id in self._active_calls:
   490→            return self._active_calls[call_id]
   491→
   492→        session = GroupCallSession(
   493→            call_id=call_id,
   494→            group_id=group_id,
   495→            host_user_id=host_user_id,
   496→            state=CallState.JOINING,
   497→        )
   498→
   499→        # Add host participant
   500→        host = CallParticipant(
   501→            user_id=host_user_id,
   502→            is_host=True,
   503→            audio_state=MediaState.ON,
   504→            joined_at=time.time(),
   505→        )
   506→        session.add_participant(host)
   507→
   508→        # Add self
   509→        self_participant = CallParticipant(
   510→            user_id=self._local_user_id,
   511→            is_host=False,
   512→            audio_state=MediaState.ON,
   513→            joined_at=time.time(),
   514→        )
   515→        session.add_participant(self_participant)
   516→
   517→        # Create WebRTC answer
   518→        sdp_answer = await self._create_webrtc_answer(session, sdp_offer)
   519→
   520→        # Send answer to host
   521→        signal = GroupCallSignal.create_answer(
   522→            call_id=call_id,
   523→            sdp=sdp_answer,
   524→        )
   525→        await self._send_signal(host_user_id, signal)
   526→
   527→        session.state = CallState.CONNECTED
   528→        session.started_at = time.time()
   529→        self._active_calls[call_id] = session
   530→
   531→        if self._on_participant_joined:
   532→            await self._on_participant_joined(session, self_participant)
   533→        if self._on_call_state_changed:
   534→            await self._on_call_state_changed(session)
   535→
   536→        logger.info("Joined group call %s", call_id)
   537→        return session
   538→
   539→    async def leave_call(self, call_id: str, reason: str = "") -> None:
   540→        """Leave an active group call."""
   541→        session = self._active_calls.get(call_id)
   542→        if session is None:
   543→            return
   544→
   545→        session.state = CallState.LEAVING
   546→
   547→        # Notify all participants
   548→        signal = GroupCallSignal.create_leave(
   549→            call_id=call_id,
   550→            user_id=self._local_user_id,
   551→            reason=reason,
   552→        )
   553→        for participant_id in session.participants:
   554→            if participant_id != self._local_user_id:
   555→                await self._send_signal(participant_id, signal)
   556→
   557→        # Close all WebRTC connections
   558→        for participant in session.participants.values():
   559→            if participant.peer_connection:
   560→                await participant.peer_connection.close()
   561→
   562→        # Remove self from session
   563→        session.remove_participant(self._local_user_id)
   564→
   565→        # If host left and still participants, transfer host
   566→        if session.host_user_id == self._local_user_id and session.participants:
   567→            new_host_id = next(iter(session.participants))
   568→            session.host_user_id = new_host_id
   569→            session.participants[new_host_id].is_host = True
   570→            logger.info("Host transferred to %s", new_host_id[:8])
   571→        elif not session.participants:
   572→            session.state = CallState.ENDED
   573→            session.ended_at = time.time()
   574→
   575→        if session.state == CallState.ENDED:
   576→            del self._active_calls[call_id]
   577→
   578→        if self._on_call_state_changed:
   579→            await self._on_call_state_changed(session)
   580→
   581→        logger.info("Left group call %s", call_id)

   587→    async def toggle_audio(self, call_id: str) -> MediaState:
   588→        """Toggle audio on/off for the local user."""
   589→        session = self._active_calls.get(call_id)
   590→        if session is None:
   591→            return MediaState.OFF
   592→
   593→        participant = session.get_participant(self._local_user_id)
   594→        if participant is None:
   595→            return MediaState.OFF
   596→
   597→        # Toggle
   598→        if participant.audio_state == MediaState.ON:
   599→            participant.audio_state = MediaState.MUTED
   600→        else:
   601→            participant.audio_state = MediaState.ON
   602→
   603→        # Notify participants
   604→        await self._broadcast_media_update(session, participant)
   605→        return participant.audio_state
   606→
   607→    async def toggle_video(self, call_id: str) -> MediaState:
   608→        """Toggle video on/off for the local user."""
   609→        session = self._active_calls.get(call_id)
   610→        if session is None:
   611→            return MediaState.OFF
   612→
   613→        participant = session.get_participant(self._local_user_id)
   614→        if participant is None:
   615→            return MediaState.OFF
   616→
   617→        if participant.video_state == MediaState.ON:
   618→            participant.video_state = MediaState.OFF
   619→        else:
   620→            participant.video_state = MediaState.ON
   621→
   622→        await self._broadcast_media_update(session, participant)
   623→        return participant.video_state
   624→
   625→    async def toggle_screen_share(self, call_id: str) -> bool:
   626→        """Toggle screen sharing for the local user."""
   627→        session = self._active_calls.get(call_id)
   628→        if session is None:
   629→            return False
   630→
   631→        participant = session.get_participant(self._local_user_id)
   632→        if participant is None:
   633→            return False
   634→
   635→        # Only one screen share at a time
   636→        if not participant.screen_share and session.is_screen_sharing:
   637→            raise GroupCallError("Another participant is already sharing their screen.")
   638→
   639→        participant.screen_share = not participant.screen_share
   640→        session.is_screen_sharing = participant.screen_share
   641→        session.screen_sharer_id = self._local_user_id if participant.screen_share else ""
   642→
   643→        await self._broadcast_media_update(session, participant)
   644→        return participant.screen_share
   645→
   646→    async def raise_hand(self, call_id: str, raised: bool = True) -> None:
   647→        """Raise or lower hand."""
   648→        session = self._active_calls.get(call_id)
   649→        if session is None:
   650→            return
   651→
   652→        participant = session.get_participant(self._local_user_id)
   653→        if participant:
   654→            participant.is_raising_hand = raised
   655→
   656→        signal = GroupCallSignal.create_raise_hand(call_id, self._local_user_id, raised)
   657→        for peer_id in session.participants:
   658→            if peer_id != self._local_user_id:
   659→                await self._send_signal(peer_id, signal)
   660→

   669→            call_id: Active call ID.

   675→    def get_active_speaker(self, call_id: str) -> Optional[str]:
   676→        """Get the user_id of the current active speaker."""

   679→        return max(self._audio_levels, key=self._audio_levels.get)  # type: ignore[arg-type]

   681→    # -------------------------------------------------------------------
   682→    # Signaling handler
   683→    # -------------------------------------------------------------------
   684→
   685→    async def handle_signal(self, sender_id: str, signal: dict[str, Any]) -> None:
   686→        """Handle an incoming group call signal."""
   687→        signal_type = signal.get("type", "")
   688→        call_id = signal.get("call_id", "")
   689→
   690→        if signal_type == "GROUP_CALL_OFFER":
   691→            await self.join_call(
   692→                call_id=call_id,
   693→                group_id=signal.get("group_id", ""),
   694→                host_user_id=signal.get("host_user_id", ""),
   695→                sdp_offer=signal.get("sdp", ""),
   696→            )
   697→
   698→        elif signal_type == "GROUP_CALL_ANSWER":
   699→            session = self._active_calls.get(call_id)
   700→            if session:
   701→                participant = session.get_participant(sender_id)
   702→                if participant:
   703→                    participant.audio_state = MediaState.ON
   704→                    if self._on_participant_joined:
   705→                        await self._on_participant_joined(session, participant)
   706→
   707→        elif signal_type == "GROUP_CALL_LEAVE":
   708→            session = self._active_calls.get(call_id)
   709→            if session:
   710→                removed = session.remove_participant(sender_id)
   711→                if removed and self._on_participant_left:
   712→                    await self._on_participant_left(session, sender_id)
   713→
   714→        elif signal_type == "GROUP_CALL_MEDIA_UPDATE":
   715→            session = self._active_calls.get(call_id)
   716→            if session:
   717→                participant = session.get_participant(sender_id)
   718→                if participant:
   719→                    participant.audio_state = MediaState(signal.get("audio", 0))
   720→                    participant.video_state = MediaState(signal.get("video", 0))
   721→                    participant.screen_share = signal.get("screen_share", False)
   722→                    if self._on_media_changed:
   723→                        await self._on_media_changed(session, participant)
   724→
   725→        elif signal_type == "GROUP_CALL_RAISE_HAND":
   726→            session = self._active_calls.get(call_id)
   727→            if session:
   728→                participant = session.get_participant(sender_id)
   729→                if participant:
   730→                    participant.is_raising_hand = signal.get("raised", False)
   731→
   732→        elif signal_type == "GROUP_CALL_ICE_CANDIDATE":
   733→            # Forward ICE candidate to the appropriate peer connection
   734→            pass  # WebRTC handles this
   735→

   737→    # Internal helpers
   738→    # -------------------------------------------------------------------
   739→
   740→    async def _broadcast_media_update(
   741→        self, session: GroupCallSession, participant: CallParticipant
   742→    ) -> None:
   743→        """Broadcast media state change to all participants."""
   744→        signal = GroupCallSignal.create_media_update(
   745→            call_id=session.call_id,
   746→            user_id=participant.user_id,
   747→            audio=participant.audio_state,
   748→            video=participant.video_state,
   749→            screen_share=participant.screen_share,
   750→        )
   751→        for peer_id in session.participants:
   752→            if peer_id != self._local_user_id:
   753→                await self._send_signal(peer_id, signal)
   754→
   755→        if self._on_media_changed:
   756→            await self._on_media_changed(session, participant)
   757→
   758→    async def _send_signal(self, peer_id: str, signal: dict[str, Any]) -> None:
   759→        """Send a signaling message to a peer."""
   760→        if self._signaling_callback:
   761→            try:
   762→                await self._signaling_callback(peer_id, signal)
   763→            except Exception as exc:
   764→                logger.error("Failed to send signal to %s: %s", peer_id[:8], exc)
